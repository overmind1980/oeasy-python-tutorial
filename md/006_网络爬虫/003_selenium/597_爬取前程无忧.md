```
import time

from selenium.webdriver.common.keys import Keys
from selenium.webdriver.common.by import By
from selenium.webdriver.firefox.service import Service
from selenium import webdriver
import pandas as pd

service = Service(executable_path="geckodriver")
driver = webdriver.Firefox(service=service)
url = "https://movie.douban.com/top250"
movies = []
for page in range(0, 250, 25):
    cur_url = f"{url}?start={page}"
    print("cur_url====",cur_url)
    driver.get(cur_url)
    time.sleep(2)
    items = driver.find_elements(By.CLASS_NAME, 'item')
    for item in items:
        title = item.find_element(By.CLASS_NAME, 'title').text
        rating = item.find_element(By.CLASS_NAME, 'rating_num').text
        info = item.find_element(By.CLASS_NAME, 'bd').text.split('\n')[1]
        movies.append({
            '标题': title,
            '评分': rating,
            '信息': info
        })
    print('\n',movies)

# 保存数据
df = pd.DataFrame(movies)
df.to_csv('./movies.csv', index=False, encoding='utf-8-sig')
driver.quit()
```

### 爬取豆瓣小组

```
import time

from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.firefox.service import Service

import pandas as pd

service = Service(executable_path="geckodriver")
driver = webdriver.Firefox(service=service)
url = "https://www.douban.com/group/698743/"  # 替换为豆瓣小组 URL
driver.get(url)
time.sleep(3)

# 登录豆瓣（如果需要登录，可手动登录后继续运行）
print("请在浏览器中完成登录（如有需要），然后按 Enter 继续...")
input()  # 等待手动登录完成

# 爬取数据
posts = []
max_posts = 200  # 目标爬取的帖子数量
page_number = 1
more_group_discussions_clicked = False  # 标记是否已点击“更多小组讨论”

while len(posts) < max_posts:
    try:
        print(f"正在爬取第 {page_number} 页...")
        # 如果尚未点击“更多小组讨论”，点击它
        if not more_group_discussions_clicked:
            try:
                more_group_discussions_button = WebDriverWait(driver, 10).until(
                    EC.element_to_be_clickable((By.LINK_TEXT, "更多小组讨论"))
                )
                more_group_discussions_button.click()
                time.sleep(3)  # 等待页面加载
                more_group_discussions_clicked = True
                print("已点击 '更多小组讨论' 按钮，开始爬取下一部分...")
            except Exception as e:
                print("未找到 '更多小组讨论' 按钮或点击失败：", e)
                break

        # 获取当前页的帖子
        post_elements = driver.find_elements(By.CLASS_NAME, "title")
        for post in post_elements:
            try:
                # 获取标题和链接
                title = post.text.strip()
                link = post.find_element(By.TAG_NAME, "a").get_attribute("href")

                # 打开帖子详情页
                driver.execute_script("window.open(arguments[0]);", link)
                driver.switch_to.window(driver.window_handles[-1])
                time.sleep(2)

                # 获取点赞量、转发量
                try:
                    like_count = driver.find_element(By.CLASS_NAME, "react-num").text.strip()
                except:
                    like_count = "0"  # 没有点赞量则为 0

                try:
                    comment_count = driver.find_element(By.CLASS_NAME, "rec-num").text.strip()
                except:
                    comment_count = "0"  # 没有转发量则为 0


                # print帖子信息
                print(title,link,like_count,comment_count,sep=",")

                # 关闭帖子详情页
                driver.close()
                driver.switch_to.window(driver.window_handles[0])
            except Exception as e:
                print(f"解析帖子失败：{e}")

        # 检查是否可以翻页
        try:
            current_url = driver.current_url
            next_button = WebDriverWait(driver, 10).until(
                EC.element_to_be_clickable((By.CSS_SELECTOR, ".next a"))
            )
            next_button.click()
            time.sleep(3)  # 等待页面加载
            if driver.current_url == current_url:
                print("翻页无效，可能已到最后一页")
                break
            page_number += 1
        except Exception as e:
            print(f"翻页失败：{e}")
            break
    except Exception as e:
        print(f"爬取第 {page_number} 页失败：{e}")
        break

# 关闭浏览器
driver.quit()
```
