### 爬取微博数据

```
import os
import time
import pickle

from selenium import webdriver
from selenium.webdriver.common.by import By


class WBCrawler(object):
    def __init__(self, word):
        # 初始化WebDriver，这里使用Edge浏览器
        if os.name == "nt":
            self.browser = webdriver.Chrome()
        else:
            self.browser = webdriver.Safari()

        # 爬取的话题
        self.search = word
        # 爬取的页面链接
        self.url = f"https://m.weibo.cn/search?containerid=231522type%3D1%26t%3D10%26q%3D%23{self.search}%23"

    def getPage(self):
        # 打开页面
        self.browser.get(self.url)
        # 等待页面加载
        time.sleep(2)

        print("页面已打开")

    def parsePage(self):
        while True:
            # 模拟滚动到页面底部
            self.browser.execute_script("window.scrollTo(0,document.body.scrollHeight)")
            # 获取微博内容
            content = self.browser.find_elements(
                By.XPATH, ".//div[@class='weibo-text']"
            )
            # print(content)
            # 获取至少50条微博数据
            if len(content) > 50:
                print(f"共获取到{len(content)}条数据")
                break
        # 转换为列表
        content_list = []
        for c in content:
            # print(c.text)
            content_list.append(c.text)
        # 将列表转换为字符串，简单进行数据处理
        content = "\n".join([line.replace("\n", "") for line in content_list]).replace(
            " ", ""
        )
        # 写入文件
        output_folder = "output"
        os.makedirs(output_folder, exist_ok=True)
        with open(
            f"output/{self.search}TopicContent.txt",
            "w",
            newline="",
            encoding="utf-8",
            errors="ignore",
        ) as f:
            f.write(content)

        # 写入数据库
        with open(f"output/{self.search}TopicContent.db", "wb") as f:
            pickle.dump(content_list, f)

        self.browser.quit()
        print("页面已关闭")
        print(f'页面已解析完成，内容已写入文件"output/{self.search}TopicContent.txt"')

    def main(self):
        # 打开页面
        self.getPage()
        # 解析页面
        self.parsePage()


if __name__ == "__main__":
    # topic = input("请输入爬取的话题：")
    topic = "中国传媒大学"
    crawler = WBCrawler(topic)
    crawler.main()
```

import jieba
import jieba.posseg as pseg
import pandas as pd
import snownlp


# 判断是否非中文且非英文
def not_chinese_or_english(word):
    for char in word:
        if ('\u4e00' <= char <= '\u9fa5') or (u'\u0041' <= char <= u'\u005a') or (u'\u0061' <= char <= u'\u007a'):
            pass
        else:
            return True
    return False


class Process:
    def __init__(self, word):
        self.topic = word
        with open(f'output/{self.topic}TopicContent.txt', 'r', newline='', encoding='utf-8') as f:
            self.text = f.read()
            f.seek(0)   # 将文件指针重新定位到文件开头
            self.lines = f.readlines()
        self.word_list = []
        self.pos_list = []
        self.count = {}
        self.sentimentCount = {}

    def sentiment_analysis(self):
        for line in self.lines:
            # print(line)
            sentiment = snownlp.SnowNLP(line).sentiments
            # print(sentiment)
            if sentiment > 0.6:
                self.sentimentCount['正面'] = self.sentimentCount.get('正面', 0) + 1
                # print('正面')
            elif sentiment < 0.4:
                self.sentimentCount['负面'] = self.sentimentCount.get('负面', 0) + 1
                # print('负面')
            else:
                self.sentimentCount['中立'] = self.sentimentCount.get('中立', 0) + 1
                # print('中立')
        print(self.sentimentCount)

        print('情感分析统计已完成')

    def segment(self):
        # 替换特殊字符,清除emoji表情
        for ch in '!"#$%&()*+,-./:;<=>?@[\\]^_‘{|}~，。？！；：”“’‘-=《》（）、【】':
            text = self.text.replace(ch, " ")
        # 分词并标注词性
        jieba.load_userdict('user_dict.txt')
        words_list_jieba = pseg.lcut(text)
        # 去除停用词
        for word, pos in words_list_jieba:
            if word in ["你", "这", "那", "有", "不", "在", "却", "都", "也", "等", "让", "又",
                        "就", "是", "就是", "我", "他", "它", "他们", "我们", "啦", "中", "会",
                        "它们", "的", "着", "了", "和", "与", "个", "一个", "一", "来", "从",
                        "呀", "及", "们",
                        " ", "～", "\n"] or not_chinese_or_english(word):
                pass
            else:
                self.word_list.append(word)
                self.pos_list.append(pos)

        dataframe = pd.DataFrame({'word': self.word_list, 'pos': self.pos_list})
        dataframe.to_csv(f'output/{self.topic}WordPos.csv', index=False, sep=',', encoding='gbk')

        print('分词与词性标注已完成，内容已写入文件"output/word-pos.csv"')

    def frequency(self):
        # 统计词频
        for key in self.word_list:
            self.count[key] = self.count.get(key, 0) + 1
        print(self.count)

        print('词频统计已完成')

import wordcloud
from matplotlib import pyplot as plt

from matplotlib import font_manager


class Visualization(object):
    def __init__(self, word):
        self.topic = word

    def drawWordCloud(self, data_dict):
        w = wordcloud.WordCloud(font_path="msyh.ttc", width=1000, height=700, background_color="white")
        w.fit_words(data_dict)
        img_path = f'output/{self.topic}WordCloud.png'

        print("绘制词云成功")
        w.to_file(img_path)
        plt.imshow(w.to_image())
        plt.axis('off')
        plt.show(block=False)
        plt.pause(5)
        plt.close()

    def drawBar(self, data_dict, labels):
        x = list(data_dict.keys())
        y = list(data_dict.values())
        title = labels[0]
        xLabel = labels[1]
        yLabel = labels[2]

        fm: font_manager.FontManager = font_manager.fontManager
        fm.addfont("msyh.ttc")
        plt.rcParams['font.sans-serif'] = ['Microsoft YaHei']
        plt.rcParams['axes.unicode_minus'] = False

        plt.figure(figsize=(8, 6))
        bars = plt.bar(x, y, color=['blue', 'red', 'green'])  # 可以根据需要指定颜色
        plt.xlabel(xLabel)
        plt.ylabel(yLabel)
        plt.title(title)
        for bar, value in zip(bars, y):
            plt.text(bar.get_x() + bar.get_width() / 2, bar.get_height(), value, ha='center', va='bottom')

        print("绘制柱状图成功")
        plt.savefig(f'output/{self.topic}{title}Hist.png')
        plt.show(block=False)
        plt.pause(5)
        plt.close()
