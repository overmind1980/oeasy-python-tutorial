import time

import csv
import os

row_dict = {}
csv_file = 'hotel.csv'
writer = None


def init_csv():
    global writer, row_dict
    if os.path.exists(csv_file):
        with open(csv_file, 'r', encoding='utf-8') as f:
            reader = csv.reader(f)
            for row in list(reader)[1:]:      # 排除行头
                if not row:
                    continue
                row = [col.strip() for col in row]   # 爬取的时候发现录入的信息会空一行，所以后面加的
                row_dict[row[0]] = tuple(row)

    f = open(csv_file, 'w', encoding='utf-8', newline='')
    writer = csv.writer(f)
    writer.writerow(['酒店名称', '酒店位置', '评分', '评分人数', '价格'])


# 初始化浏览器
def init_browser():
    option = Options()
    option.add_experimental_option('excludeSwitches', ['enable-automation'])
    option.add_argument('--disable-blink-features=AutomationControlled')
    # 本地谷歌浏览器路径
    # option._binary_location = r'C:\Users\Administrator\AppData\Local\Google\Chrome\Bin\chromex.exe'
    # service = Service(executable_path=ChromeDriverManager().install())
    # driver = webdriver.Chrome(service=service, options=option)
    driver = webdriver.Chrome(executable_path=ChromeDriverManager().install(), options=option)
    return driver

# 扫码登录，只需第一次扫
def load_cookies(driver):
    driver.get('https://www.ctrip.com/')
    with open('cookies.json', 'r', encoding='utf-8') as f:
        cookies = json.load(f)
        for cookie in cookies:
            driver.add_cookie(cookie)


def save_cookies(cookies):
    with open('cookies.json', 'w', encoding='utf-8') as f:
        json.dump(cookies, f, indent=4)


# 登录
def login(driver):
    login_url = 'https://passport.ctrip.com/user/login'
    driver.get(login_url)
    # 扫码登录按钮
    ele_scan = driver.find_element(By.XPATH, '/html/body/div[1]/div[2]/div[2]/div[2]/div/div/div[1]/div[5]/a')
    ele_scan.click()
    time.sleep(2)


# 遍历获取酒店列表
def get_hotel_info(driver):
    global row_dict
    lis_xpath = r'//*[@id="ibu_hotel_container"]/div/section/div[2]/ul/li'
    list_li = driver.find_elements(By.XPATH, lis_xpath)
    for i, ele_li in enumerate(list_li):
        try:
            li_class = ele_li.get_attribute('class')
            # 过滤不是酒店项的元素节点
            if li_class != 'list-item-target':
                continue
            # 获取酒店名称
            name = ele_li.find_element(By.XPATH, './div/div/div/div[1]/div[2]/div[1]/div/span').text.strip()
            # print(f'{name=}')
            if name in row_dict:
                continue

            # 获取酒店位置
            loc = ele_li.find_element(By.XPATH, './div/div/div/div[1]/div[2]/div[2]/p/span[1]/span[1]').text
            # 评分
            score = ele_li.find_element(By.XPATH, './div/div/div/div[2]/div[1]/div/div[2]/span').text
            # input(f'{score=}')
            # 评分人数
            score_num = ele_li.find_element(By.XPATH, './div/div/div/div[2]/div[1]/div/div[1]/p[2]/a').text
            score_num = score_num.replace('条点评', '').replace(',', '')
            # 价格
            price_xpath = './div/div/div/div[2]/div[2]/div[1]/p/span[2]/span[{}]'
            price = ele_li.find_element(By.XPATH, price_xpath.format(2)).text
            if price.strip() == '起':        # 如果是起，说明该酒店没有打折，则取第一个span元素
                price = ele_li.find_element(By.XPATH, price_xpath.format(1)).text
            price = price.replace('¥', '').replace(',', '')
            print('酒店信息', name, loc, score, score_num, price)
            row_dict[name] = (name, loc, score, score_num, price)
        except Exception as e:
            print(f'酒店报错:{e}')
            continue


# 检查扫码登录是否成功
def check_scan_login_status(driver):
    flag = False
    scan_url = 'https://passport.ctrip.com/user/login'
    while not flag:
        print('等待扫码登录...')
        try:
            assert not driver.current_url.startswith(scan_url)  # 如果当前页面还是扫码页面，则报错
            flag = True
        except:
            time.sleep(2)
    print('登录成功')
    time.sleep(2)
    cookies = driver.get_cookies()
    save_cookies(cookies)


def main():
    global writer, row_dict
    init_csv()
    driver = init_browser()
    if not os.path.exists('cookies.json'):
        login(driver)
        check_scan_login_status(driver)
    else:
        load_cookies(driver)

    # 打开搜索页面
    list_url = 'https://hotels.ctrip.com/hotels/list?countryId=1&city=1&provinceId=0&checkin=2024/08/01&checkout=2024/08/07&optionId=1&optionType=City&directSearch=0&display=%E5%8C%97%E4%BA%AC%2C%20%E4%B8%AD%E5%9B%BD&crn=1&adult=1&children=0&searchBoxArg=t&travelPurpose=0&ctm_ref=ix_sb_dl&domestic=1&'
    driver.get(list_url)

    # 窗口比例缩放为50%
    driver.execute_script("document.body.style.zoom = '80%'")

    time.sleep(3)
    if driver.current_url.startswith('https://passport.ctrip.com/user/login'):
        print('登录cookies已失效，请删除cookies.json文件后再重新爬虫扫码登录')
        return

    # 要爬的页数
    page = 60
    for i in range(page):
        get_hotel_info(driver)
        print('第' + str(i+1) + '页结束')

        # 窗口比例缩放为50%
        time.sleep(1)
        driver.execute_script("document.body.style.zoom = '50%'")
        time.sleep(1)

        # 滚动到页面底部
        driver.execute_script("window.scrollTo(0, document.body.scrollHeight);")
        time.sleep(3)
        # 如果翻到下一页后存在【搜索更多酒店】的按钮,则点击
        btn_search_xpath = '/html/body/div[2]/div[2]/div/div/section/div[2]/ul/div[2]/div'
        try:
            if EC.presence_of_element_located((By.XPATH, btn_search_xpath)):
                element = driver.find_element(By.XPATH, btn_search_xpath)
                driver.execute_script("arguments[0].click();", element)
                time.sleep(3)
        except Exception as e:
            print(f'第{i+1}页翻到下一页报错失败')
            time.sleep(1)
            driver.execute_script("document.body.style.zoom = '100%'")
            time.sleep(1)
            continue
        time.sleep(1)
        driver.execute_script("document.body.style.zoom = '100%'")
        time.sleep(1)

    if '' in row_dict:
        del row_dict['']
    # 写入数据
    writer.writerows(list(row_dict.values()))
    print('爬虫结束')


if __name__ == '__main__':
    main()
