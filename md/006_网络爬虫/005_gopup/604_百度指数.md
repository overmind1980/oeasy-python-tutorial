### 多个关键字指数

```
import gopup as gp
cookie = """具体的cookie"""
df1 = gp.baidu_search_index(word="口罩", start_date='2020-01-01', end_date='2022-12-01', cookie=cookie)
print(df1)
df2 = gp.baidu_search_index(word="疫情", start_date='2020-01-01', end_date='2022-12-01', cookie=cookie)
l = list(zip(df1.index,df1.values,df2.values))
for data in l:
    #print(data)
    day = str(data[0]).replace(" 00:00:00","")
    index1 = data[1][2]
    index2 = data[2][2]
    print(day,",",index1,",",index2)
```
