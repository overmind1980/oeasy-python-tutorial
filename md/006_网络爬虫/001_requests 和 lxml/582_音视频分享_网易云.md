def process_text(comments):
    # 合并所有评论为一个字符串
    text = ' '.join(comments)
    # 使用jieba进行分词
    word_list = jieba.lcut(text)
    # 去除长度小于2的词语

    return word_list

# 定义函数：绘制评论数量随时间变化的折线图
def plot_comments_over_time(comments):
    comment_dates = [datetime.fromtimestamp(comment['time']/1000) for comment in comments]
    dates_sorted = sorted(comment_dates)
    date_counts = {}
    for date in dates_sorted:
        date_str = date.strftime('%Y-%m-%d')
        if date_str in date_counts:
            date_counts[date_str] += 1
        else:
            date_counts[date_str] = 1
    
    sorted_dates = sorted(date_counts.items())

    x = [datetime.strptime(date[0], '%Y-%m-%d') for date in sorted_dates]
    y = [count[1] for count in sorted_dates]

    fig, ax = plt.subplots()
    ax.xaxis.set_major_formatter(DateFormatter('%Y-%m-%d'))
    ax.plot(x, y, marker='o', linestyle='-', color='b')
    ax.set_xlabel('Date')
    ax.set_ylabel('Number of Comments')
    ax.set_title('Number of Comments Over Time')

    plt.xticks(rotation=45)
    plt.tight_layout()
    plt.show()

import os
import csv
import requests
import re


def get_song_info(song_id):
    url = f"https://music.163.com/api/song/detail/?ids=[{song_id}]"
    print(f"访问URL: {url}")

    try:
        response = requests.get(url)
        response.raise_for_status()
        data = response.json()

        # 提取歌曲题目
        song_title = data.get('songs', [{}])[0].get('name', None)
        if not song_title:
            print("未能获取歌曲题目")
            return None

        return song_title

    except Exception as e:
        print(f"获取歌曲题目失败: {e}")
        return None


def get_lyrics(song_id):
    url = f"https://music.163.com/api/song/lyric?id={song_id}&lv=1&kv=1&tv=-1"
    print(f"访问URL: {url}")

    try:
        response = requests.get(url)
        response.raise_for_status()
        data = response.json()

        # 提取歌词数据
        lyrics = data.get('lrc', {}).get('lyric', None)
        if not lyrics:
            print("未能获取歌词")
            return None

        # 清理歌词内容
        # 去除时间戳
        lyrics_cleaned = re.sub(r'\[.*?\]', '', lyrics)
        # 去除作词人、作曲人的信息
        lyrics_cleaned = re.sub(r'作词\s*:\s*\S+\n', '', lyrics_cleaned)
        lyrics_cleaned = re.sub(r'作曲\s*:\s*\S+\n', '', lyrics_cleaned)
        lyrics_cleaned = lyrics_cleaned.strip()

        return lyrics_cleaned

    except Exception as e:
        print(f"获取歌词失败: {e}")
        return None


def save_lyrics_to_csv(song_ids, filename):
    desktop_path = os.path.join(os.path.expanduser("~"), "Desktop")
    full_path = os.path.join(desktop_path, filename)

    with open(full_path, 'w', newline='', encoding='utf-8') as csvfile:
        writer = csv.writer(csvfile)
        writer.writerow(['Song ID', 'Song Title', 'Lyrics'])

        for song_id in song_ids:
            song_title = get_song_info(song_id)
            lyrics = get_lyrics(song_id)
            if song_title and lyrics:
                print(f"成功获取歌曲: {song_title}")
                print(f"成功获取歌词: {lyrics}")
                writer.writerow([song_id, song_title, lyrics])
            else:
                print(f"未能获取歌曲信息或歌词，Song ID: {song_id}")
                writer.writerow(
                    [song_id, song_title if song_title else '未能获取歌曲题目', lyrics if lyrics else '未能获取歌词'])

    print(f"歌曲信息和歌词已保存到 {full_path}")


def main():
    song_ids = ["574925512", "1487202784", "1471111588", "2052381087", "1960130296", "536623692", "2051707056",
                "1486106184", "574925509"]
    filename = "lyrics.csv"
    save_lyrics_to_csv(song_ids, filename)


if __name__ == "__main__":
    main()
