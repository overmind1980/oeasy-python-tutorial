import requests
import re
import jieba
from wordcloud import WordCloud
import matplotlib.pyplot as plt
from snownlp import SnowNLP
import pandas as pd


def get_danmaku(bvid):
    """
    根据给定的BV号获取视频的弹幕内容。

    参数:
    bvid (str): Bilibili视频的BV号

    返回:
    list: 弹幕内容的列表
    """
    # 添加请求头以模拟浏览器访问
    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36',
        'Referer': f'https://www.bilibili.com/video/{bvid}'
    }

    # 获取视频信息的URL
    info_url = f"https://api.bilibili.com/x/web-interface/view?bvid={bvid}"
    response = requests.get(info_url, headers=headers)

    if response.status_code != 200:
        print(f"请求失败，状态码：{response.status_code}")
        return []

    response_text = response.json()

    try:
        # 从响应中提取视频的cid
        cid = response_text['data']['cid']
    except KeyError:
        print("未能找到cid，请检查BV号是否正确或API响应格式是否变化")
        return []

    # 获取弹幕XML内容的URL
    danmaku_url = f"https://comment.bilibili.com/{cid}.xml"

    # 获取弹幕XML内容
    response = requests.get(danmaku_url, headers=headers)
    response.encoding = 'utf-8'
    danmaku_xml = response.text

    # 使用正则表达式提取弹幕内容
    danmaku_list = re.findall(r'<d p=".*?">(.*?)</d>', danmaku_xml)

    return danmaku_list


def generate_wordcloud(danmaku_list):
    """
    根据弹幕内容生成词云图。

    参数:
    danmaku_list (list): 弹幕内容的列表
    """
    if not danmaku_list:
        print("没有弹幕数据，无法生成词云图")
        return

    # 将弹幕列表拼接成一个字符串
    text = ' '.join(danmaku_list)

    # 使用jieba库对文本进行分词
    wordlist = jieba.cut(text, cut_all=False)
    word_space_split = " ".join(wordlist)

    # 生成词云图
    wordcloud = WordCloud(font_path='msyh.ttc', width=800, height=400, background_color='white').generate(
        word_space_split)

    # 显示词云图
    plt.figure(figsize=(10, 5))
    plt.imshow(wordcloud, interpolation='bilinear')
    plt.axis('off')
    plt.show()


def sentiment_analysis(danmaku_list):
    """
    对弹幕内容进行情感分析，并计算平均情感值和中位情感值。

    参数:
    danmaku_list (list): 弹幕内容的列表
    """
    if not danmaku_list:
        print("没有弹幕数据，无法进行情感分析")
        return

    # 使用SnowNLP库对每条弹幕进行情感分析
    sentiments = [SnowNLP(danmaku).sentiments for danmaku in danmaku_list]
    sentiment_series = pd.Series(sentiments)

    # 计算平均情感值和中位情感值
    sentiment_mean = sentiment_series.mean()
    sentiment_median = sentiment_series.median()

    print(f"平均情感值: {sentiment_mean:.2f}")
    print(f"中位情感值: {sentiment_median:.2f}")


if __name__ == "__main__":
    # Bilibili视频的BV号
    bvid = 'BV12i421S7iH'

    # 获取弹幕内容
    danmaku_list = get_danmaku(bvid)

    print("生成词云图...")
    generate_wordcloud(danmaku_list)

    print("进行情感分析...")
    sentiment_analysis(danmaku_list)

import requests
import xml.etree.ElementTree as ET
import pandas as pd
from datetime import datetime


def get_cid(bv_id):
    video_info_url = f'https://api.bilibili.com/x/web-interface/view?bvid={bv_id}'
    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
        'Accept-Language': 'zh-CN,zh;q=0.9',
        'Referer': 'https://www.bilibili.com/',
        'Connection': 'keep-alive',
    }

    try:
        response = requests.get(video_info_url, headers=headers, timeout=10)
        response.raise_for_status()
        data = response.json()

        if 'data' in data and 'cid' in data['data']:
            return data['data']['cid']
        else:
            print("未能获取到有效的CID")
            return None
    except requests.RequestException as e:
        print("请求出错：", e)
        return None


def format_time(seconds):
    minutes = int(seconds // 60)
    seconds = int(seconds % 60)
    return f"{minutes:02d}:{seconds:02d}"


def format_timestamp(unix_timestamp):
    return datetime.fromtimestamp(unix_timestamp).strftime('%Y-%m-%d %H:%M:%S')


def fetch_bilibili_danmaku(bv_id, output_base='关税弹幕'):  # 改为基础文件名
    cid = get_cid(bv_id)
    if not cid:
        print("无法获取CID，跳过")
        return

    danmaku_api_url = f'https://api.bilibili.com/x/v1/dm/list.so?oid={cid}'
    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
        'Accept-Language': 'zh-CN,zh;q=0.9',
        'Referer': 'https://www.bilibili.com/',
        'Connection': 'keep-alive',
    }

    try:
        response = requests.get(danmaku_api_url, headers=headers, timeout=10)
        response.raise_for_status()
        response.encoding = 'utf-8'

        root = ET.fromstring(response.content)
        danmakus = []

        for dm in root.findall('d'):
            p = dm.get('p').split(',')
            danmakus.append({
                '弹幕时间': format_time(float(p[0])),
                '模式': int(p[1]),
                '字体大小': int(p[2]),
                '颜色': int(p[3]),
                '发送时间': format_timestamp(int(p[4])),
                '弹幕池': int(p[5]),
                '发送者ID': p[6],
                '行ID': p[7],
                '内容': dm.text.strip() if dm.text else '无内容'
            })

        if danmakus:
            df = pd.DataFrame(danmakus)
            
            # 生成两种格式的文件名
            xlsx_file = f"{output_base}.xlsx"
            csv_file = f"{output_base}.csv"
            
            # 保存Excel
            df.to_excel(xlsx_file, index=False)
            # 保存CSV
            df.to_csv(csv_file, index=False, encoding='utf-8-sig')
            
            print(f"数据已保存至:\n{xlsx_file}\n{csv_file}")
        else:
            print("未找到弹幕数据")

    except requests.RequestException as e:
        print("请求出错：", e)
    except ET.ParseError:
        print("解析弹幕XML数据时出错")


# 示例调用
bv_id = "BV1eF3tzjESn"  # 请替换为目标视频的BV号
fetch_bilibili_danmaku(bv_id)
