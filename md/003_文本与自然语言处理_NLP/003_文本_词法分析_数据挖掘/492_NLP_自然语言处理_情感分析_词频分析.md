```
import nltk
from nltk.sentiment import SentimentIntensityAnalyzer

# Download necessary resources (only required once)
nltk.download('vader_lexicon')

# Create an instance of the SentimentIntensityAnalyzer
sia = SentimentIntensityAnalyzer()

# Input text for sentiment analysis
text = "I am feeling happy today!"

# Perform sentiment analysis
sentiment_scores = sia.polarity_scores(text)

# Interpret the sentiment scores
if sentiment_scores['compound'] >= 0.05:
    sentiment = "Positive"
elif sentiment_scores['compound'] <= -0.05:
    sentiment = "Negative"
else:
    sentiment = "Neutral"

# Print the sentiment and scores
print("Sentiment: ", sentiment)
print("Sentiment Scores: ", sentiment_scores)
```

### snownlp 情感分析

```
import jieba
from snownlp import SnowNLP
from collections import Counter

def get_text():
    text = input("请输入一段中文文本：")
    return text

def get_num_topics():
    num_topics = int(input("请输入需要归纳的主题数："))
    return num_topics

def tokenize_text(text):
    tokens = jieba.lcut(text)
    return tokens

def extract_keywords(tokens, num_topics):
    counter = Counter(tokens)
    top_keywords = [word for word, _ in counter.most_common(num_topics)]
    return top_keywords

def analyze_sentiment(text):
    s = SnowNLP(text)
    sentiment = s.sentiments
    return "正面" if sentiment > 0.5 else "负面" if sentiment < 0.5 else "中性"


if __name__ == "__main__":
    text = get_text()
    num_topics = get_num_topics()
    tokens = tokenize_text(text)
    print("tokens", tokens)
    keywords = extract_keywords(tokens, num_topics)
    print("提取的主题关键词：")
    for idx, keyword in enumerate(keywords):
        print(f"关键词 {idx + 1}: {keyword}")
    sentiment = analyze_sentiment(text)
    print(f"整体情感倾向: {sentiment}")
```

```
import pandas as pd
import re
import jieba
from snownlp import SnowNLP
import matplotlib.pyplot as plt
from matplotlib.font_manager import FontProperties

# 读取CSV文件
try:
    data = pd.read_csv('jiangping.csv', encoding='GBK')
except UnicodeDecodeError:
    data = pd.read_csv('jiangping.csv', encoding='utf-8')

# 假设评论在第二列（列名为 '评论'）
comments = data['评论'].dropna().tolist()

# 加载中文停用词表
def load_stopwords(filepath):
    encodings = ['utf-8', 'GBK', 'ISO-8859-1']
    for enc in encodings:
        try:
            with open(filepath, 'r', encoding=enc) as file:
                stopwords = file.read().splitlines()
            return stopwords
        except UnicodeDecodeError:
            continue
    raise ValueError(f"Could not decode file {filepath} with available encodings.")

# 停用词文件
stopwords_filepath = '/Users/fancyxu/Desktop/jisuanchuanbo/pythonProject1/chinese_stopwords.txt'
stopwords = load_stopwords(stopwords_filepath)

# 自定义无意义词语列表
meaningless_words = ['的', '了', '是', '我', '也', '很', '和', '就', '不', '在', '有', '人', '都', '一']

# 将无意义词语添加到停用词列表中
stopwords.extend(meaningless_words)

# 文本清洗和分词函数
def clean_and_tokenize(text):
    text = re.sub('[^\u4e00-\u9fa5a-zA-Z0-9\s]', '', text)  # 去除标点符号
    words = jieba.lcut(text)  # 使用jieba进行分词
    words = [word for word in words if word not in stopwords and len(word) > 1]  # 去除停用词和单个字符
    return words

# 清洗和分词所有评论
tokenized_comments = [clean_and_tokenize(comment) for comment in comments]

# 扁平化列表
all_words = [word for sublist in tokenized_comments for word in sublist]

# 情感分析
positive_count = 0
angry_count = 0
anxious_count = 0

for comment in comments:
    s = SnowNLP(comment)
    sentiments_score = s.sentiments  # 得到情感得分，范围是0（负面）到1（正面）
    if sentiments_score > 0.6:
        positive_count += 1
    elif 0.4 < sentiments_score <= 0.6:
        anxious_count += 1
    else:
        angry_count += 1

# 统计结果
labels = ['积极', '气愤', '焦虑']
sizes = [positive_count, angry_count, anxious_count]
colors = ['#ff9999','#66b3ff','#99ff99']
explode = (0.1, 0, 0)  # 使积极情绪那一块稍微突出

# 设置字体属性
font_path = '/System/Library/Fonts/PingFang.ttc'
font_properties = FontProperties(fname=font_path)

# 绘制饼状图
plt.figure(figsize=(8, 8))
plt.pie(sizes, explode=explode, labels=labels, colors=colors, autopct='%1.1f%%',
        shadow=True, startangle=140, textprops={'fontproperties': font_properties})
plt.axis('equal')  # 保证饼状图是一个正圆
plt.title('情感倾向分布', fontproperties=font_properties)
plt.show()
```

### textblob

```
import jieba
from textblob import TextBlob
from wordcloud import WordCloud

def segment_text(text):
    '''分词'''
    words = ' '.join(jieba.cut(text))
    return words

def analyze_sentiment(text):
    '''情感分析'''
    blob = TextBlob(text)
    sentiment_score = blob.sentiment.polarity
    sentiment_type = "Positive" if sentiment_score > 0 else "Negative" if sentiment_score < 0 else "Neutral"
    return sentiment_score, sentiment_type

s = "今天天气不错"
sentiment = analyze_sentiment(s)
print(sentiment)
```
