import jieba

from collections import Counter

from wordcloud import WordCloud

import matplotlib.pyplot as plt

import string

import re





def generate_keywords_summary(text):

    words = []  # 初始化一个空列表



    # 将英文标点符号替换为空格

    text = text.translate(str.maketrans(string.punctuation, ' ' * len(string.punctuation)))



    # 使用正则表达式去除中文标点符号和其他特殊字符

    text = re.sub(r'[^\w\s]', '', text)



    words = jieba.lcut(text)  # 中文分词



    # 去除空格和停用词

    stopwords = []

    with open('C:/Users/DELL/Desktop/stopwords.txt', 'r', encoding='utf-8') as f:

        stopwords = [line.strip() for line in f]



    cleaned_words = [word.replace('\n', '').replace('\u3000', '') for word in words]

    cleaned_words = [word for word in cleaned_words if word not in stopwords]



    word_counts = Counter(cleaned_words)  # 统计词频



    # 获取出现频率最高的5个词作为关键词总结

    keywords = [word for word, count in word_counts.most_common(5)]



    return keywords





# 从文件中读取新闻文本

file_path = 'C:/Users/DELL/Desktop/测试.txt'  # 替换为目标新闻文件路径



with open(file_path, 'r', encoding='utf-8') as file:

    news_text = file.read()



keywords_summary = generate_keywords_summary(news_text)

print("关键词总结:", keywords_summary)



# 设置字体

font_path = 'C:/windows/fonts/SimHei.ttf'  # 替换为你的字体文件路径



# 构建关键词云图

wordcloud = WordCloud(font_path=font_path, width=1920, height=1080, background_color='white').generate(news_text)



# 显示关键词云图

fig, ax = plt.subplots()

ax.imshow(wordcloud, interpolation='bilinear')

ax.axis('off')



# 添加标题

plt.title('wordcloud')



# 显示图形

plt.show()
