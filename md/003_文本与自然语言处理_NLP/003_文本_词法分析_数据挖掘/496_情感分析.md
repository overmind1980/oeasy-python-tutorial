```
def analyze_sentiment(text):  
    # 创建一个TextBlob对象  
    blob = TextBlob(text)  
      
    # 计算情感分析结果  
    result = blob.sentiment  
      
    # 返回情感极性（0-1之间，表示负面到正面的情感）和情感强度（1-4之间，表示较弱到强烈的情感）  
    return result.polarity, result.subjectivity  
  
# 测试函数
```

# Plutchik 情绪轮 Python 实现示例
Plutchik 情绪轮（Plutchik's Wheel of Emotions）是心理学经典的**8种基础情绪**模型，包含：喜悦、信任、恐惧、惊讶、悲伤、厌恶、愤怒、期待。

下面给你**两个最实用的 Python 实现**：
1. 用 **Matplotlib** 绘制标准 Plutchik 情绪轮（可视化）
2. 用 **字典/分类** 实现情绪匹配（文本/数据处理）

## 二、完整代码：绘制 Plutchik 情绪轮
```python
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.patches import Wedge, Circle

# ===================== 1. 定义 Plutchik 8 种基础情绪 =====================
emotions = [
    "Joy\n喜悦", "Trust\n信任", "Fear\n恐惧", 
    "Surprise\n惊讶", "Sadness\n悲伤", "Disgust\n厌恶", 
    "Anger\n愤怒", "Anticipation\n期待"
]

# 对应颜色（Plutchik 标准配色）
colors = [
    "#FCEA2B",  # 喜悦-黄
    "#54D9EB",  # 信任-青
    "#9F81FF",  # 恐惧-紫
    "#FF91E7",  # 惊讶-粉
    "#4A90E2",  # 悲伤-蓝
    "#70D669",  # 厌恶-绿
    "#FF5252",  # 愤怒-红
    "#FFB142"   # 期待-橙
]

# ===================== 2. 绘制环形情绪轮 =====================
fig, ax = plt.subplots(figsize=(10, 10), subplot_kw={'projection': 'polar'})
ax.set_ylim(0, 3)
ax.set_xticks([])
ax.set_yticks([])
ax.set_theta_zero_location("N")  # 0° 指向正北
ax.set_theta_direction(-1)       # 顺时针绘制

angle_step = 2 * np.pi / 8  # 每个情绪 45°

# 绘制 8 个情绪扇形
for i in range(8):
    theta1 = i * angle_step
    theta2 = (i + 1) * angle_step
    
    # 外层大扇形
    wedge = Wedge((0, 0), 3, np.degrees(theta1), np.degrees(theta2), 
                  color=colors[i], alpha=0.8)
    ax.add_patch(wedge)
    
    # 添加情绪文字
    angle = (theta1 + theta2) / 2
    x = 2.2 * np.cos(angle)
    y = 2.2 * np.sin(angle)
    ax.text(x, y, emotions[i], ha='center', va='center', fontsize=12, fontweight='bold')

# 中心白色圆
center_circle = Circle((0, 0), 0.5, color='white')
ax.add_patch(center_circle)

plt.title("Plutchik's Wheel of Emotions\n普鲁奇克情绪轮", fontsize=16, pad=20)
plt.tight_layout()
plt.show()
```

### 运行效果
会生成一个**标准 8 色 Plutchik 情绪轮**，包含中文+英文标注，可直接用于论文、报告、数据分析可视化。

```python
# Plutchik 8 大基础情绪分类字典
plutchik_emotions = {
    "joy": ["喜悦", "开心", "快乐", "幸福", "满足", "兴奋", "愉悦"],
    "trust": ["信任", "接纳", "认可", "欣赏", "接受", "安心"],
    "fear": ["恐惧", "害怕", "担忧", "焦虑", "紧张", "不安", "恐慌"],
    "surprise": ["惊讶", "意外", "吃惊", "震惊", "诧异"],
    "sadness": ["悲伤", "难过", "失落", "沮丧", "伤心", "忧郁"],
    "disgust": ["厌恶", "反感", "讨厌", "嫌弃", "恶心"],
    "anger": ["愤怒", "生气", "恼怒", "暴躁", "发火", "不满"],
    "anticipation": ["期待", "期望", "盼望", "憧憬", "准备"]
}

# 示例：判断一句话的情绪
def get_plutchik_emotion(text):
    for emotion, keywords in plutchik_emotions.items():
        for kw in keywords:
            if kw in text:
                return emotion, keywords[0]
    return "unknown", "未知"

# 测试
text = "我很害怕明天的考试"
emotion_en, emotion_cn = get_plutchik_emotion(text)
print(f"文本：{text}")
print(f"匹配 Plutchik 情绪：{emotion_en} ({emotion_cn})")
```

## 四、进阶：复合情绪（Plutchik 核心理论）
Plutchik 认为**两种基础情绪可以组合成高级情绪**，比如：
- 喜悦 + 信任 = 爱
- 恐惧 + 惊讶 = 敬畏
- 愤怒 + 厌恶 = 鄙视
