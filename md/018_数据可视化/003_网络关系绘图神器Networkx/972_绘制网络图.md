### 引入

代码解释这是一个关于小社群内谣言传播的代码。描述了在10个人中，从一个人开始，向若干人散播一个可信度为50％的谣言，然后相信的人再向若干人散播这个谣言（成功率为50%）满足以下两种条件，程序将终止并呈现结果：1、最后的谣言传播者即将传播的人已经相信谣言2、所有人相信（必然会停止）在程序终止后，将输出最后谣言的相信者编号。这个模型很接近于传播学专业的舆情分析模型，只是本人做了极大的简化，其中不免漏洞重重。但这个谣言小范围传播依然能较真实地反映一些传播现象。

```
import networkx as nx  
import random   
def create_random_graph(num_people):  
    G = nx.Graph()  
    G.add_nodes_from(range(num_people))      
    for i in G.nodes():  
        connected_nodes = random.sample(G.nodes(), random.randint(1, num_people-1))  
        connected_nodes = [n for n in connected_nodes if n != i]  
        G.add_edges_from([(i, n) for n in connected_nodes])  
    return G  
  
def spread_rumor(G, start_person, prob_spread=0.5):   
    rumor_spread_set = set([start_person])    
    current_spread_set = set([start_person])        
    while current_spread_set:  
        new_spread_set = set()  
        for person in current_spread_set:  
            neighbors = G.neighbors(person)  
            for neighbor in neighbors:  
                if neighbor not in rumor_spread_set:                    if random.random() < prob_spread:  
                        rumor_spread_set.add(neighbor)  
                        new_spread_set.add(neighbor)  
        current_spread_set = new_spread_set  
return rumor_spread_set  
  
num_people = 10  
G = create_random_graph(num_people)  
start_person = 0  
spread_result = spread_rumor(G, start_person)  
print("谣言传到的人:", sorted(spread_result))
```

import pandas as pd
import jieba
import networkx as nx
import matplotlib.pyplot as plt
from collections import Counter
from itertools import combinations
import re

# 读取 CSV 文件
def read_csv(file_path):
    df = pd.read_csv(file_path)
    return df

# 提取文本中的高频词汇
def extract_high_freq_words(text, num_words=20):
    stop_words = set()  
    with open('chinese_stopwords.txt', 'r', encoding='utf-8') as f:
        stop_words.update(f.read().split())

    words = jieba.lcut(text)
    filtered_words = [word for word in words if word not in stop_words and len(word) > 1 and not re.match(r'^\d+$', word) and '#' not in word]
    word_freq = Counter(filtered_words)
    return word_freq.most_common(num_words)

# 构建语义网络图
def build_semantic_network(word_freq):
    G = nx.Graph()
    for word, freq in word_freq:
        G.add_node(word, size=freq)

    # 添加一些边以展示关联
    words = [word for word, freq in word_freq]
    for word_pair in combinations(words, 2):
        G.add_edge(*word_pair)

    return G

# 绘制语义网络图
def draw_network(G):
    pos = nx.spring_layout(G)
    sizes = [G.nodes[node]['size'] * 4 for node in G.nodes]
    
    plt.figure(figsize=(12, 8))
    nx.draw_networkx_nodes(G, pos, node_size=sizes, node_color='skyblue', alpha=0.5)
    nx.draw_networkx_edges(G, pos, edge_color='gray', alpha=0.3)
    nx.draw_networkx_labels(G, pos, font_size=10, font_family='SimHei', font_weight='bold')

    plt.title('语义网络分析', fontsize=15, fontfamily='SimHei')
    plt.axis('off')
    plt.show()

# 主函数
def main(file_path):
    df = read_csv(file_path)
    text = ' '.join(df['word']) 
    word_freq = extract_high_freq_words(text)
    G = build_semantic_network(word_freq)
    draw_network(G)

# 调用主函数
file_path = '1.csv'  
main(file_path)

```


### 基于PMI的语义共现网络分析

import pandas as pd
import numpy as np
import networkx as nx
from itertools import combinations
from collections import Counter

# 读取分词后的Excel文档
df = pd.read_excel(r'C:\Users\meihaoweilai\Desktop\爬取抖音数据\分词结果1.xlsx')

# 获取分词后的文本数据
text_data = df['text_cut'].tolist()

# 处理非字符串类型的值
text_data = [str(text) if not isinstance(text, str) else text for text in text_data]

# 计算词频
word_freq = Counter(word for text in text_data for word in text.split())

# 计算共现矩阵
co_occurrence = Counter()
for text in text_data:
    unique_words = set(text.split())
    for word1, word2 in combinations(unique_words, 2):
        co_occurrence[(word1, word2)] += 1

# 计算PMI
pmi = {}
N = sum(word_freq.values())
for pair, freq in co_occurrence.items():
    pmi[pair] = np.log2((freq * N) / (word_freq[pair[0]] * word_freq[pair[1]]))

# 选取PMI值最高的前3000个词对
top_3000_pairs = sorted(pmi.items(), key=lambda x: x[1], reverse=True)[:3000]

# 创建图
G = nx.Graph()

# 添加节点和边
for pair, value in top_3000_pairs:
    source, target = pair
    G.add_node(source)
    G.add_node(target)
    G.add_edge(source, target, weight=value)

# 导出为GEXF文件
nx.write_gexf(G, r'C:\Users\o\Desktop\爬取抖音数据\基于PMI的共现网络图.gexf')

text_data = [text for text in text_data if isinstance(text, str)]

word_freq = Counter(word for text in text_data for word in text.split())

co_occurrence = Counter()
for text in text_data:
    unique_words = set(text.split())
    for word1, word2 in combinations(unique_words, 2):
        co_occurrence[word1, word2] += 1
        co_occurrence[word2, word1] += 1

# 计算PMI
pmi = {}
N = sum(word_freq.values())
for pair, freq in co_occurrence.items():
    pmi[pair] = np.log2((freq * N) / (word_freq[pair[0]] * word_freq[pair[1]]))

# 设定共现频次的阈值
frequency_threshold = 3

# 选取PMI值最高的前3000个词对，但同时要求共现频次超过阈值
filtered_pairs = [(pair, pmi_val) for pair, pmi_val in pmi.items() if co_occurrence[pair] > frequency_threshold]
top_3000_pairs = sorted(filtered_pairs, key=lambda x: x[1], reverse=True)[:3000]

# 创建图
G = nx.Graph()

# 添加节点和边
for pair, value in top_3000_pairs:
    source, target = pair
    G.add_node(source)
    G.add_node(target)
    G.add_edge(source, target, weight=value)

# 导出为GEXF文件

### 红楼梦人物

```
import networkx as nx
import matplotlib.pyplot as plt
import jieba
import re
from collections import defaultdict

plt.rcParams['font.sans-serif'] = ['WenQuanYi Zen Hei']
plt.rcParams['axes.unicode_minus'] = False

def load_characters():
    # 主要人物列表
    characters = {
        '宝玉': ['贾宝玉', '二爷'],
        '黛玉': ['林黛玉', '林妹妹'],
        '宝钗': ['薛宝钗', '薛姑娘'],
        '凤姐': ['王熙凤', '凤姐儿'],
        '袭人': ['袭人'],
        '晴雯': ['晴雯'],
        '探春': ['探春'],
        '湘云': ['史湘云'],
        '紫鹃': ['紫鹃'],
        '贾母': ['贾母', '老太太'],
        '王夫人': ['王夫人'],
        '贾政': ['贾政'],
        '贾琏': ['贾琏'],
        '贾环': ['贾环'],
    }
    return characters

def build_relationship_network(chapters):
    characters = load_characters()
    relationships = defaultdict(lambda: defaultdict(int))
    
    # 遍历每个章节
    for chapter in chapters:
        content = chapter['content']
        # 在同一段落中出现的人物可能有关系
        paragraphs = content.split('\n')
        
        for para in paragraphs:
            appeared_chars = set()
            # 检查每个人物是否在段落中出现
            for main_char, aliases in characters.items():
                for alias in aliases:
                    if alias in para:
                        appeared_chars.add(main_char)
                        break
            
            # 建立共现关系
            appeared_chars = list(appeared_chars)
            for i in range(len(appeared_chars)):
                for j in range(i + 1, len(appeared_chars)):
                    char1, char2 = appeared_chars[i], appeared_chars[j]
                    relationships[char1][char2] += 1
                    relationships[char2][char1] += 1
    
    return relationships

def draw_network(relationships):
    G = nx.Graph()
    
    # 添加边和权重
    for char1, related in relationships.items():
        for char2, weight in related.items():
            if weight > 50:  # 过滤弱关系
                G.add_edge(char1, char2, weight=weight)
    
    plt.figure(figsize=(15, 15))
    pos = nx.spring_layout(G, k=1, iterations=50)
    
    # 绘制边，粗细表示关系强度
    edge_weights = [G[u][v]['weight'] for u, v in G.edges()]
    nx.draw_networkx_edges(G, pos, width=[w/50 for w in edge_weights], 
                          edge_color='gray', alpha=0.5)
    
    # 绘制节点和标签
    nx.draw_networkx_nodes(G, pos, node_size=2000, node_color='lightblue')
    nx.draw_networkx_labels(G, pos, font_size=12)
    
    plt.title('红楼梦人物关系网络')
    plt.axis('off')
    plt.show()

def analyze_relationships(filename):
    with open(filename, 'r', encoding='utf-8') as f:
        content = f.read()
    
    pattern = r'第[零一二三四五六七八九十百]+回\s+[\u4e00-\u9fa5]+[\u4e00-\u9fa5\s]+[\u4e00-\u9fa5]'
    chapters = re.split(pattern, content)
    titles = re.findall(pattern, content)
    
    chapter_list = []
    for title, content in zip(titles, chapters[1:]):
        chapter_list.append({
            'title': title.strip(),
            'content': content.strip()
        })
    
    relationships = build_relationship_network(chapter_list)
    draw_network(relationships)
    return relationships

if __name__ == "__main__":
    relationships = analyze_relationships('hlm.txt')
