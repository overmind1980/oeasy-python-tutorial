
- 两只老虎变奏曲


```
import sys

import mido

TICKS_PER_BEAT = 480
CHANNEL = 0
PROGRAM = 0
VELOCITY = 100

MELODY = [
    [60, 1],
    [62, 1],
    [64, 1],
    [60, 1],
    [60, 1],
    [62, 1],
    [64, 1],
    [60, 1],
]

TEMPO_CHANGES = [
    [0, 90],
    [4, 180],
]

mid = mido.MidiFile(type=0, ticks_per_beat=TICKS_PER_BEAT)
track = mido.MidiTrack()
mid.tracks.append(track)

current_tick = 0
note_events: list[tuple[int, int, mido.Message]] = []
for pitch, beats in MELODY:
    duration_ticks = int(beats) * TICKS_PER_BEAT
    note_events.append((current_tick, 1, mido.Message("note_on", channel=CHANNEL, note=int(pitch), velocity=VELOCITY, time=0)))
    note_events.append((current_tick + duration_ticks, 2, mido.Message("note_off", channel=CHANNEL, note=int(pitch), velocity=0, time=0)))
    current_tick += duration_ticks

tempo_events: list[tuple[int, int, mido.Message]] = []
tempo_events.append((0, 0, mido.Message("program_change", channel=CHANNEL, program=PROGRAM, time=0)))
for beat, bpm in TEMPO_CHANGES:
    tick = int(beat) * TICKS_PER_BEAT
    tempo_events.append((tick, -1, mido.MetaMessage("set_tempo", tempo=mido.bpm2tempo(int(bpm)), time=0)))
tempo_events.append((current_tick, 9, mido.MetaMessage("end_of_track", time=0)))

events = note_events + tempo_events
events.sort(key=lambda x: (x[0], x[1]))

last_tick = 0
for tick, _, msg in events:
    delta = tick - last_tick
    track.append(msg.copy(time=delta))
    last_tick = tick

out = sys.argv[1] if len(sys.argv) > 1 else "two_tigers_1track_tempo.mid"
mid.save(out)
print(out)
```
