### 快乐的情绪

```
from midiutil import MIDIFile
import random

# 创建MIDI对象（4个轨道：主旋律、和声、低音、打击乐）
midi = MIDIFile(4)
melody_track = 0
harmony_track = 1
bass_track = 2
drum_track = 3
tempo = 130  # 欢快的速度

# 设置音量
volumes = {
    'melody': 100,
    'harmony': 80,
    'bass': 85,
    'drum': 75
}

# 为每个轨道设置速度
for track in range(4):
    midi.addTempo(track, 0, tempo)

# C大调音阶音符
scale = [60, 62, 64, 65, 67, 69, 71, 72]  # C4到C5
chord_roots = [60, 65, 67, 62]  # C, F, G, D

# 生成主题动机（4拍）
def create_motif():
    motif = []
    # 使用更简单的固定节奏模式
    rhythm = [(1, 1, 1, 1), (1, 1, 2), (0.5, 0.5, 1, 2)]  # 确保每个模式总和为4拍
    pattern = random.choice(rhythm)
    
    for duration in pattern:
        note = random.choice(scale)
        motif.append((note, duration))
    return motif

# 生成变奏动机
def vary_motif(motif):
    varied = []
    for note, duration in motif:
        if random.random() < 0.3:  # 30%概率改变音高
            note = note + random.choice([-2, 2, 4])
        varied.append((note, duration))
    return varied

# 生成完整的主旋律
melody = []
main_motif = create_motif()
total_bars = 16  # 添加总小节数定义

# AABA形式
for section in range(4):
    if section != 2:  # A段
        melody.extend(main_motif)
        melody.extend(vary_motif(main_motif))
    else:  # B段（对比部分）
        contrast_motif = create_motif()
        melody.extend(contrast_motif)
        melody.extend(vary_motif(contrast_motif))

# 添加主旋律
current_time = 0
for note, duration in melody:
    # 确保音符在音阶范围内
    note = max(min(note, 74), 60)
    midi.addNote(melody_track, 0, note, current_time, duration, volumes['melody'])
    current_time += duration

# 生成和声进行
harmony_progression = []
for _ in range(8):  # 修改为固定8个和弦进行
    root = random.choice(chord_roots)
    # 大三和弦：根音、三度、五度
    chord = [(root, 4), (root + 4, 4), (root + 7, 4)]
    harmony_progression.extend(chord)

# 添加和声
current_time = 0
for note, duration in harmony_progression:
    midi.addNote(harmony_track, 1, note, current_time, duration, volumes['harmony'])
    current_time += duration

# 生成贝斯线
current_time = 0
for _ in range(total_bars):
    root = random.choice(chord_roots)
    # 每小节的贝斯模式：根音-五度-根音-三度
    pattern = [(root-24, 1), (root-24+7, 1), (root-24, 1), (root-24+4, 1)]
    for note, duration in pattern:
        midi.addNote(bass_track, 2, note, current_time, duration, volumes['bass'])
        current_time += duration

# 添加打击乐
current_time = 0
while current_time < total_bars * 4:
    # 底鼓
    midi.addNote(drum_track, 9, 35, current_time, 0.5, volumes['drum'])
    midi.addNote(drum_track, 9, 35, current_time + 2, 0.5, volumes['drum'])
    
    # 军鼓
    midi.addNote(drum_track, 9, 38, current_time + 1, 0.5, volumes['drum'])
    midi.addNote(drum_track, 9, 38, current_time + 3, 0.5, volumes['drum'])
    
    # 击镲
    for i in range(4):
        midi.addNote(drum_track, 9, 42, current_time + i, 0.25, volumes['drum'] - 20)
    
    current_time += 4

# 保存MIDI文件
try:
    with open("happy_random.mid", "wb") as output_file:
        midi.writeFile(output_file)
    print("欢快的随机曲子已生成完成！")
except Exception as e:
    print(f"生成MIDI文件时出错：{e}")
    print("请确保目录存在且有写入权限")
```

### 悲伤的情绪

```
from midiutil import MIDIFile
import random

# 创建MIDI对象（4个轨道）
midi = MIDIFile(4)
melody_track = 0
harmony_track = 1
bass_track = 2
piano_track = 3
tempo = 72  # 较慢的速度，表达忧伤

# 设置音量
volumes = {
    'melody': 100,    # 主旋律
    'harmony': 70,    # 和声
    'bass': 80,       # 低音
    'piano': 85       # 钢琴伴奏
}

# 为每个轨道设置速度
for track in range(4):
    midi.addTempo(track, 0, tempo)

# A小调音阶音符
scale = [69, 71, 72, 74, 76, 77, 79, 81]  # A4到A5
chord_roots = [69, 74, 76, 71]  # Am, Dm, Em, Bm

# 生成忧伤的主题动机（4拍）
def create_sad_motif():
    motif = []
    # 较长的音符，表达悲伤情绪
    rhythm = [(2, 2), (3, 1), (1.5, 1.5, 1)]
    pattern = random.choice(rhythm)
    
    for duration in pattern:
        note = random.choice(scale)
        motif.append((note, duration))
    return motif

# 生成变奏
def vary_sad_motif(motif):
    varied = []
    for note, duration in motif:
        if random.random() < 0.4:
            note = note - random.choice([2, 3])  # 倾向于下行
        varied.append((note, duration))
    return varied

# 生成主旋律
melody = []
main_motif = create_sad_motif()
total_bars = 16

for section in range(4):
    melody.extend(main_motif)
    melody.extend(vary_sad_motif(main_motif))

# 添加主旋律
current_time = 0
for note, duration in melody:
    note = max(min(note, 81), 69)
    midi.addNote(melody_track, 0, note, current_time, duration, volumes['melody'])
    current_time += duration

# 创建MIDI对象（5个轨道）
midi = MIDIFile(5)
melody_track = 0
violin_track = 1
harmony_track = 2
bass_track = 3
piano_track = 4

# 设置音量
volumes = {
    'melody': 100,    # 主旋律
    'violin': 95,     # 小提琴
    'harmony': 70,    # 和声
    'bass': 80,       # 低音
    'piano': 85       # 钢琴伴奏
}

# 为每个轨道设置速度
for track in range(5):
    midi.addTempo(track, 0, tempo)

# 添加小提琴持续音
def create_violin_line():
    violin_notes = []
    for _ in range(total_bars * 2):
        base_note = random.choice(scale)
        # 生成长音符
        duration = random.choice([2, 3, 4])
        # 添加颤音效果
        for i in range(int(duration * 2)):
            if random.random() < 0.3:
                violin_notes.append((base_note + 1, 0.25))
                violin_notes.append((base_note, 0.25))
            else:
                violin_notes.append((base_note, 0.5))
    return violin_notes

# 添加小提琴声部
violin_line = create_violin_line()
current_time = 0
for note, duration in violin_line:
    midi.addNote(violin_track, 0, note, current_time, duration, volumes['violin'])
    current_time += duration

# 添加小调和弦进行
harmony_progression = [
    [(69, 72, 76), 4],  # Am
    [(74, 77, 81), 4],  # Dm
    [(76, 79, 83), 4],  # Em
    [(71, 74, 78), 4],  # Bm
]

current_time = 0
for chord, duration in harmony_progression * 4:
    for note in chord:
        midi.addNote(harmony_track, 1, note, current_time, duration, volumes['harmony'])
    current_time += duration

# 添加深沉的贝斯线
current_time = 0
for _ in range(total_bars):
    root = random.choice(chord_roots)
    pattern = [
        (root-24, 2),
        (root-24+7, 1),
        (root-24+3, 1)
    ]
    for note, duration in pattern:
        midi.addNote(bass_track, 2, note, current_time, duration, volumes['bass'])
    current_time += 4

# 添加钢琴装饰音
current_time = 0
while current_time < total_bars * 4:
    if random.random() < 0.7:  # 70%概率添加装饰音
        note = random.choice(scale) - 12
        midi.addNote(piano_track, 3, note, current_time, 0.5, volumes['piano'] - 20)
    current_time += 1

# 保存MIDI文件
try:
    with open("sad_random.mid", "wb") as output_file:
        midi.writeFile(output_file)
    print("悲伤的随机曲子已生成完成！")
except Exception as e:
    print(f"生成MIDI文件时出错：{e}")
    print("请确保目录存在且有写入权限")
```
