from midiutil import MIDIFile
import random

class DetailedJRockGenerator:
    def __init__(self):
        self.midi = MIDIFile(4)
        self.bpm = 160

        # 定义具体的主旋律音符序列（使用MIDI音符编号）
        self.main_melody = {
            # A段主题
            "A": [
                # 第1小节
                (72, 1.0, 100),  # C5, 四分音符, 力度100
                (74, 0.5, 95),   # D5, 八分音符
                (76, 0.5, 95),   # E5
                (77, 1.0, 90),   # F5
                (79, 1.0, 100),  # G5

                # 第2小节
                (77, 0.5, 90),   # F5
                (76, 0.5, 85),   # E5
                (74, 1.0, 85),   # D5
                (72, 2.0, 95),   # C5

                # 第3-4小节（重复变奏）
                (74, 1.0, 100),  # D5
                (76, 0.5, 95),   # E5
                (77, 0.5, 95),   # F5
                (79, 2.0, 100),  # G5
                (76, 1.0, 90),   # E5
                (74, 1.0, 85),   # D5
                (72, 2.0, 95),   # C5
            ],

            # B段（副歌）
            "B": [
                # 第1小节
                (79, 1.0, 110),  # G5
                (81, 0.5, 105),  # A5
                (83, 0.5, 105),  # B5
                (84, 2.0, 110),  # C6

                # 第2小节
                (83, 1.0, 100),  # B5
                (81, 1.0, 95),   # A5
                (79, 2.0, 100),  # G5

                # 第3-4小节
                (81, 1.0, 105),  # A5
                (79, 0.5, 100),  # G5
                (77, 0.5, 95),   # F5
                (76, 2.0, 100),  # E5
                (74, 1.0, 90),   # D5
                (72, 1.0, 85),   # C5
                (71, 2.0, 95),   # B4
            ]
        }

        # 定义具体的和弦进行
        self.chord_sequence = [
            # A段和弦进行
            ([60,64,67], 4),     # C大三和弦
            ([67,71,74], 4),     # G大三和弦
            ([65,69,72], 4),     # F大三和弦
            ([62,65,69], 4),     # Dm三和弦

            # B段和弦进行
            ([60,64,67], 4),     # C大三和弦
            ([65,69,72], 4),     # F大三和弦
            ([67,71,74], 4),     # G大三和弦
            ([60,64,67], 4),     # C大三和弦
        ]

        # 定义贝斯线
        self.bass_line = [
            # A段贝斯
            (48, 1.0, 100),  # C3
            (48, 1.0, 90),   # C3
            (55, 1.0, 95),   # G3
            (48, 1.0, 90),   # C3

            # 重复
            (48, 1.0, 100),  # C3
            (50, 1.0, 90),   # D3
            (52, 1.0, 95),   # E3
            (53, 1.0, 90),   # F3
        ]

        # 初始化所有轨道
        for track in range(4):
            self.midi.addTempo(track, 0, self.bpm)
            self.midi.addProgramChange(track, 0, 0, self._get_instrument(track))

    def _get_instrument(self, track):
        instruments = {
            0: 5,    # 电吉他（主旋律）
            1: 30,   # 失真电吉他（和弦）
            2: 33,   # 电贝斯
            3: 0     # 标准鼓组
        }
        return instruments[track]

    def generate_melody(self):
        track = 0
        time = 0

        # 生成完整曲目结构：前奏-A-A-B-A-B-A-尾奏
        structure = ["A", "A", "B", "A", "B", "A"]

        for section in structure:
            for note, duration, velocity in self.main_melody[section]:
                self.midi.addNote(track, 0, note, time, duration, velocity)

                # 添加装饰音（滑音效果）
                if random.random() < 0.2:
                    self.midi.addPitchWheelEvent(track, 0, time, 8192 + 2048)
                    self.midi.addPitchWheelEvent(track, 0, time + 0.1, 8192)

                time += duration

    def generate_chords(self):
        track = 1
        time = 0

        # 重复和弦进行
        for _ in range(6):  # 6个段落
            for chord, duration in self.chord_sequence:
                for note in chord:
                    self.midi.addNote(track, 0, note, time, duration, 75)
                time += duration

    def generate_bass(self):
        track = 2
        time = 0

        # 重复贝斯线
        for _ in range(24):  # 24个小节
            for note, duration, velocity in self.bass_line:
                self.midi.addNote(track, 0, note, time, duration, velocity)
                time += duration

    def generate_drums(self):
        track = 3
        time = 0

        # 定义鼓点
        kick = 36    # 底鼓
        snare = 38   # 军鼓
        hihat = 42   # 闭合镲
        crash = 49   # 碎音镲

        # 生成96个小节的鼓点
        for bar in range(96):
            # 每8小节添加碎音镲
            if bar % 8 == 0:
                self.midi.addNote(track, 9, crash, time, 1, 100)

            # 每小节的鼓点模式
            for beat in range(4):
                # 底鼓（1和3拍）
                if beat in [0, 2]:
                    self.midi.addNote(track, 9, kick, time, 1, 100)

                # 军鼓（2和4拍）
                if beat in [1, 3]:
                    self.midi.addNote(track, 9, snare, time, 1, 90)

                # 十六分音符闭合镲
                for i in range(4):
                    self.midi.addNote(track, 9, hihat, time + i/4, 0.25, 70)

                time += 1

    def generate_song(self):
        self.generate_drums()
        self.generate_bass()
        self.generate_chords()
        self.generate_melody()

        # 保存MIDI文件
        with open("detailed_jrock.mid", "wb") as output_file:
            self.midi.writeFile(output_file)

if __name__ == "__main__":
    generator = DetailedJRockGenerator()
    generator.generate_song()
    print("MIDI文件已生成：detailed_jrock.mid")

from mido import MidiFile, MidiTrack, Message, MetaMessage

# 创建 MIDI 文件
midi = MidiFile(ticks_per_beat=480)

track0 = MidiTrack()
midi.tracks.append(track0)
track0.append(MetaMessage('track_name', name='Tempo Track', time=0))

                drum_notes.append((38, beat_time, 0.1))
            drum_notes.append((42, beat_time, 0.1))

current_time = 0
for pitch, start, duration in drum_notes:

    duration_ticks = int(duration * 480)
    delta_time = start_ticks - current_time
    if delta_time < 0:
        delta_time = 0
    track2.append(Message('note_on', note=pitch, velocity=90, channel=9, time=delta_time))
    current_time = start_ticks
    track2.append(Message('note_off', note=pitch, velocity=0, channel=9, time=duration_ticks))
    current_time += duration_ticks

track2.append(MetaMessage('end_of_track', time=0))

# 保存到桌面

filename = os.path.join(desktop, "rock_style_30s.mid")

print(f"MIDI 文件已保存到 {filename}")
