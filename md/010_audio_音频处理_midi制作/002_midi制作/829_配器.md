import mido
from mido import Message, MidiFile, MidiTrack

# 配置参数
TICKS_PER_BEAT = 960  # 高分辨率
TEMPO = 500000        # 每分钟120拍 (微秒/拍)

# 初始化MIDI文件
mid = MidiFile(ticks_per_beat=TICKS_PER_BEAT)
mid.add_track('Conductor')  # 用于存放全局控制信息

# 增强轨道创建函数
def create_track(program, channel, name):
    track = MidiTrack()
    track.append(Message('program_change', program=program, time=0, channel=channel))
    track.name = name
    track.channel = channel
    track.time_counter = 0  # 时间计数器
    return track

# 创建完整交响乐团声部
strings = [
    create_track(40, 0, 'Violins I'),    # 小提琴1
    create_track(40, 1, 'Violins II'),   # 小提琴2
    create_track(42, 2, 'Violas'),       # 中提琴
    create_track(43, 3, 'Cellos'),       # 大提琴
    create_track(44, 4, 'Double Basses') # 低音提琴
]

woodwinds = [
    create_track(73, 5, 'Flute'),        # 长笛
    create_track(68, 6, 'Oboe'),         # 双簧管
    create_track(71, 7, 'Clarinets'),    # 单簧管
    create_track(70, 8, 'Bassoons')      # 巴松管
]

brass = [
    create_track(60, 9, 'French Horns'), # 圆号
    create_track(56, 10, 'Trumpets'),    # 小号
    create_track(57, 11, 'Trombones'),   # 长号
    create_track(58, 12, 'Tuba')         # 大号
]

percussion = create_track(0, 9, 'Timpani')  # 定音鼓

# 添加所有轨道
for section in [strings, woodwinds, brass, [percussion]]:
    mid.tracks.extend(section)

ode_to_joy_theme = [
    (64, 480, 70), (64, 240, 75), (65, 240, 80), (67, 480, 85),
    (67, 240, 80), (65, 240, 75), (64, 240, 70), (62, 480, 65),

]

def add_note(track, note, duration, velocity):
    """添加带时间控制的音符"""
    track.append(Message('note_on', note=note, velocity=velocity, 
                       time=track.time_counter, channel=track.channel))
    track.append(Message('note_off', note=note, velocity=velocity, 
                       time=duration, channel=track.channel))
    track.time_counter = 0  # 重置计数器

def staggered_entrance(theme, tracks, interval=480, key_shift=0):
    """声部错开进入"""
    for i, track in enumerate(tracks):
        for note_info in theme:
            note = note_info[0] + key_shift + (i*2)  # 各声部间隔三度
            duration = note_info[1]
            velocity = note_info[2] - (i*5)  # 动态渐变
            add_note(track, note, duration, velocity)
        track.time_counter += interval  # 声部间隔

def create_extended_intro():
    """扩展版引子 (3分钟)"""
    # 弦乐分层进入 (D小调分解和弦)
    chord_notes = [62, 65, 69]  # D-F-A
    for i, track in enumerate(strings):
        base_note = chord_notes[i % 3]
        for octave in range(3):
            add_note(track, base_note + (octave*12), 960, 50+i*5)
            track.time_counter += 480 if octave < 2 else 0

    # 木管延迟进入 (第30秒)
    woodwind_theme = [(60, 960, 50), (64, 480, 55), (67, 480, 60)]
    for track in woodwinds:
        track.time_counter = 1800  # 30秒后进入
        staggered_entrance(woodwind_theme, [track])

    # 定音鼓节奏动机 (每15秒出现)
    for _ in range(4):
        add_note(percussion, 60, 240, 80)
        add_note(percussion, 60, 240, 70)
        percussion.time_counter += 1440  # 间隔15秒

def develop_theme():
    """主题发展部 (5分钟)"""
    # 弦乐组卡农式进入
    canon_entries = [
        (strings[0], 0),    # 小提琴I立即进入
        (strings[1], 480),   # 小提琴II晚一拍
        (strings[3], 960),   # 大提琴晚两拍
        (strings[4], 1440)   # 低音提琴晚三拍
    ]
    for track, delay in canon_entries:
        track.time_counter = delay
        staggered_entrance(ode_to_joy_theme*2, [track], key_shift=5)

    # 铜管组辉煌插入 (第3分钟)
    brass_theme = [(60, 240, 100), (64, 480, 110), (67, 960, 120)]
    for track in brass:
        track.time_counter = 5400  # 3分钟后进入
        staggered_entrance(brass_theme*3, [track])

def grand_finale():
    """宏伟终曲 (4分钟)"""
    # 全奏和弦序列
    chord_progression = [
        (60, 64, 67),  # C大三和弦

        (64, 67, 72),  # E大三和弦
        (65, 69, 72)   # F属七和弦

        # 弦乐组

            for note in chord:
                add_note(track, note+24, 960, 90)  # 高八度
            track.time_counter = 240
        
        # 铜管组
        for track in brass:
            for note in chord:
                add_note(track, note+12, 960, 110)
            track.time_counter = 240
        
        # 木管组装饰音
        for track in woodwinds:
            add_note(track, max(chord)+3, 120, 80)
            add_note(track, max(chord)+4, 120, 75)
            track.time_counter = 240

# 构建完整结构

        'strings': pretty_midi.Instrument(program=48, name="Strings Ensemble"),
        'electric_guitar': pretty_midi.Instrument(program=27, name="Electric Guitar (Clean)"),
        'xylophone': pretty_midi.Instrument(program=14, name="Xylophone"),
        'fx_synth': pretty_midi.Instrument(program=99, name="FX Synth"),
        'special_percussion': pretty_midi.Instrument(program=118, name="Special Percussion"),
        'synth_bass': pretty_midi.Instrument(program=38, name="Synth Bass"),
        'synth_lead': pretty_midi.Instrument(program=81, name="Synth Lead"),
        'synth_pad': pretty_midi.Instrument(program=89, name="Synth Pad"),
        'synth_fx': pretty_midi.Instrument(program=103, name="Synth FX"),
        'synth_brass': pretty_midi.Instrument(program=62, name="Synth Brass"),
        'synth_pluck': pretty_midi.Instrument(program=84, name="Synth Pluck")
    }

    # 邓紫棋风格的和弦进行
    chord_progression = [
        # 主歌部分 - 更简单的和弦进行
        ('Em', 2), ('C', 2), ('G', 2), ('D', 2),
        ('Em', 2), ('C', 2), ('G', 2), ('D', 2),
        ('Em', 2), ('C', 2), ('G', 2), ('D', 2),
        ('Em', 2), ('C', 2), ('G', 2), ('D', 2),
        # 副歌部分 - 更丰富的和弦进行
        ('Em7', 2), ('Cadd9', 2), ('Gmaj7', 2), ('Dmaj7', 2),
        ('Em7', 2), ('Cadd9', 2), ('Gmaj7', 2), ('Dmaj7', 2),
        ('Em7', 2), ('Cadd9', 2), ('Gmaj7', 2), ('Dmaj7', 2),
        ('Em7', 2), ('Cadd9', 2), ('Gmaj7', 2), ('Dmaj7', 2)
    ]

    # 定义和弦对应的音名
    chord_notes_map = {
        'Em': ['E', 'G', 'B'],
        'C': ['C', 'E', 'G'],
        'G': ['G', 'B', 'D'],
        'D': ['D', 'F#', 'A'],
        'Em7': ['E', 'G', 'B', 'D'],
        'Cadd9': ['C', 'E', 'G', 'D'],
        'Gmaj7': ['G', 'B', 'D', 'F#'],
        'Dmaj7': ['D', 'F#', 'A', 'C#']
    }

    # 主歌部分的鼓点节奏 - 更简单稳定
    verse_drum_pattern = [
        (36, [0, 2]),  # 底鼓
        (38, [1, 3]),  # 军鼓
        (42, [0, 0.5, 1, 1.5, 2, 2.5, 3, 3.5]),  # 踩镲
        (51, [0, 2]),  # 吊镲
        (46, [1, 3])  # 开镲
    ]

    # 副歌部分的鼓点节奏 - 更有力量感但不会太快
    chorus_drum_pattern = [
        (36, [0, 1, 2, 3]),  # 底鼓
        (38, [0.5, 1.5, 2.5, 3.5]),  # 军鼓
        (42, [0, 0.5, 1, 1.5, 2, 2.5, 3, 3.5]),  # 踩镲
        (51, [0, 2]),  # 吊镲
        (46, [1, 3]),  # 开镲
        (45, [0.75, 2.75])  # 低音鼓
    ]

    # 生成旋律的函数
    def generate_melody(scale_notes, start_time, duration, intensity=100, is_chorus=False):
        melody = []
        num_notes = int(duration * (2 if is_chorus else 1))  # 副歌部分音符更密集
        prev_pitch = None

        for i in range(num_notes):
            if prev_pitch is not None:
                # 副歌部分允许更大的音程跳跃
                max_interval = 4 if is_chorus else 2
                candidates = [p for p in [pretty_midi.note_name_to_number(n) for n in scale_notes] if
                              abs(p - prev_pitch) <= max_interval]
                if not candidates:
                    candidates = [pretty_midi.note_name_to_number(n) for n in scale_notes]
                pitch = np.random.choice(candidates)
            else:
                pitch = pretty_midi.note_name_to_number(np.random.choice(scale_notes))
            prev_pitch = pitch

            # 副歌部分音量更大
            base_velocity = 100 if is_chorus else 80
            velocity = base_velocity + np.random.randint(-10, 10)

            # 副歌部分音符更短
            note_duration = 0.25 if is_chorus else 0.5
            note_start = start_time + i * note_duration
            note_end = note_start + note_duration

            note = pretty_midi.Note(
                velocity=velocity,
                pitch=pitch,
                start=note_start,
                end=note_end
            )
            melody.append(note)
        return melody

    current_time = 0
    for index, (chord, duration) in enumerate(chord_progression):
        is_chorus = index >= len(chord_progression) // 2

        # 钢琴和弦伴奏
        chord_notes = []
        for octave in [3, 4, 5]:
            for note_name in chord_notes_map[chord]:
                note = f'{note_name}{octave}'
                pitch = pretty_midi.note_name_to_number(note)
                chord_notes.append(pitch)

        # 副歌部分钢琴伴奏更丰富
        if is_chorus:
            for note in chord_notes:
                instruments['piano'].notes.append(pretty_midi.Note(
                    velocity=90,
                    pitch=note,
                    start=current_time,
                    end=current_time + duration
                ))
        else:
            for note in chord_notes:
                instruments['piano'].notes.append(pretty_midi.Note(
                    velocity=80,
                    pitch=note,
                    start=current_time,
                    end=current_time + duration
                ))

        # 贝斯线 - 副歌部分更活跃
        bass_note = pretty_midi.note_name_to_number(f'{chord_notes_map[chord][0]}2')
        if is_chorus:
            # 副歌部分贝斯加入切分音
            instruments['bass'].notes.append(pretty_midi.Note(
                velocity=85,
                pitch=bass_note,
                start=current_time,
                end=current_time + duration
            ))
            instruments['bass'].notes.append(pretty_midi.Note(
                velocity=75,
                pitch=bass_note,
                start=current_time + 0.5,
                end=current_time + 1
            ))
        else:
            instruments['bass'].notes.append(pretty_midi.Note(
                velocity=70,
                pitch=bass_note,
                start=current_time,
                end=current_time + duration
            ))

        # 根据段落选择鼓组节奏
        drum_pattern = chorus_drum_pattern if is_chorus else verse_drum_pattern

        # 鼓点节奏
        for drum_note, beats in drum_pattern:
            for beat in beats:
                if beat < duration:
                    instruments['drums'].notes.append(pretty_midi.Note(
                        velocity=90 if is_chorus else 80,
                        pitch=drum_note,
                        start=current_time + beat,
                        end=current_time + beat + 0.1
                    ))

        # 主音合成器旋律
        scale = {
            'Em': ['E4', 'F#4', 'G4', 'A4', 'B4', 'C5', 'D5'],
            'C': ['C4', 'D4', 'E4', 'F4', 'G4', 'A4', 'B4'],
            'G': ['G4', 'A4', 'B4', 'C5', 'D5', 'E5', 'F#5'],
            'D': ['D4', 'E4', 'F#4', 'G4', 'A4', 'B4', 'C#5']
        }.get(chord)
        if scale:
            instruments['lead_synth'].notes.extend(generate_melody(
                scale, current_time, duration, 90 if is_chorus else 80, is_chorus
            ))

        # 弦乐长音铺垫 - 副歌部分更丰富
        root_note = pretty_midi.note_name_to_number(f'{chord_notes_map[chord][0]}3')
        if is_chorus:
            for note in chord_notes:
                instruments['strings'].notes.append(pretty_midi.Note(
                    velocity=70,
                    pitch=note,
                    start=current_time,
                    end=current_time + duration
                ))
        else:
            instruments['strings'].notes.append(pretty_midi.Note(
                velocity=60,
                pitch=root_note,
                start=current_time,
                end=current_time + duration
            ))

        # 电吉他扫弦 - 副歌部分更密集
        if is_chorus or index % 2 == 0:
            for note in chord_notes:
                instruments['electric_guitar'].notes.append(pretty_midi.Note(
                    velocity=85 if is_chorus else 75,
                    pitch=note,
                    start=current_time,
                    end=current_time + 0.1
                ))

        # 合成器音效 - 副歌部分更丰富
        if is_chorus:
            for fx_note, beats in [(60, [0, 2]), (62, [1, 3]), (64, [0.5, 2.5])]:
                for beat in beats:
                    if beat < duration:
                        instruments['synth_fx'].notes.append(pretty_midi.Note(
                            velocity=80,
                            pitch=fx_note,
                            start=current_time + beat,
                            end=current_time + beat + 0.1
                        ))

        # 合成器铜管 - 主要在副歌部分
        if is_chorus and index % 2 == 0:
            for note in chord_notes:
                instruments['synth_brass'].notes.append(pretty_midi.Note(
                    velocity=75,
                    pitch=note,
                    start=current_time,
                    end=current_time + duration
                ))

        # 合成器拨弦 - 主要在副歌部分
        if is_chorus and index % 2 == 0:
            for note in chord_notes:
                instruments['synth_pluck'].notes.append(pretty_midi.Note(
                    velocity=90,
                    pitch=note,
                    start=current_time,
                    end=current_time + 0.5
                ))

        # 木琴装饰音 - 主要在副歌部分
        if is_chorus and index % 2 == 0:
            xylophone_notes = [pretty_midi.note_name_to_number(f'{chord_notes_map[chord][0]}5')]
            for note in xylophone_notes:
                instruments['xylophone'].notes.append(pretty_midi.Note(
                    velocity=90,
                    pitch=note,
                    start=current_time,
                    end=current_time + 0.5
                ))

        current_time += duration

    # 将所有乐器添加到 MIDI 对象
    for inst in instruments.values():
        midi.instruments.append(inst)

    # 获取用户主目录
    home_dir = os.path.expanduser("~")
    # 构建保存文件的完整路径
    file_path = os.path.join(home_dir, 'gem_style_song.mid')
    # 保存 MIDI 文件
    midi.write(file_path)
    print(f"MIDI 文件已保存到: {file_path}")
