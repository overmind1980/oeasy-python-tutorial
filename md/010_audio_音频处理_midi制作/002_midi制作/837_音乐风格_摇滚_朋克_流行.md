```
from midiutil import MIDIFile

# 初始化MIDI文件
midi = MIDIFile(4)  # 使用4个音轨
tempo = 128  # BPM
time_signature = (4, 4)  # 4/4拍

# 轨道配置（修改名称部分为英文）
TRACKS = {
    0: {'program': 0, 'name': 'Piano'},       # 原"钢琴主旋律"
    1: {'program': 48, 'name': 'Strings'},    # 原"弦乐和声"
    2: {'program': 33, 'name': 'Bass'},       # 原"贝斯线"
    3: {'program': 0, 'name': 'Drums'}        # 原"鼓点节奏"
}

# 初始化各轨道
for track in TRACKS:
    midi.addTrackName(track, 0, TRACKS[track]['name'])
    midi.addTempo(track, 0, tempo)
    if track != 3:  # 鼓轨不需要program change
        midi.addProgramChange(track, 0, 0, TRACKS[track]['program'])


# =====================
# 前奏部分 (8小节)
# =====================
def add_intro():
    time = 0  # 起始时间

    # 钢琴旋律
    piano_notes = [
        (60, 0, 1), (64, 1, 0.5), (67, 1.5, 0.5),  # C大调分解和弦
        (65, 2, 1), (69, 3, 0.5), (72, 3.5, 0.5),  # F大调分解
        (67, 4, 2), (72, 6, 2)  # 长音保持
    ]

    # 弦乐和声层
    string_chords = [
        ([48, 55, 60], 0, 4),  # C大三和弦
        ([53, 57, 60], 4, 4)  # F大三和弦
    ]

    # 贝斯线
    bass_line = [
        (36, 0, 1), (35, 1, 1), (33, 2, 1), (31, 3, 1),
        (36, 4, 2), (35, 6, 2)
    ]

    # 鼓点节奏
    drum_pattern = [
        (36, 0, 0.5), (42, 0.5, 0.5),  # 底鼓+踩镲
        (38, 2, 0.5), (42, 2.5, 0.5),  # 军鼓+踩镲
        (36, 4, 0.5), (42, 4.5, 0.5)  # 重复
    ]

    # 添加音符到轨道
    add_notes(0, piano_notes, time)
    add_chords(1, string_chords, time)
    add_notes(2, bass_line, time)
    add_drums(3, drum_pattern, time, repeat=4)


# =====================
# 主歌部分 (16小节)
# =====================
def add_verse():
    time = 8  # 接续前奏

    # 主旋律发展
    melody = [
        (72, 0, 1.5), (71, 1.5, 0.5), (69, 2, 1),  # 上行旋律
        (67, 3, 0.5), (65, 3.5, 0.5), (64, 4, 2),  # 下行解决
        (60, 6, 1), (64, 7, 1), (67, 8, 3)  # 高潮铺垫
    ]

    # 和弦进行
    chord_progression = [
        ([60, 64, 67], 0, 2),  # C
        ([65, 69, 72], 2, 2),  # F
        ([58, 62, 65], 4, 2),  # Dm
        ([67, 71, 74], 6, 2)  # G
    ]

    # 添加各声部
    add_notes(0, melody, time)
    add_chords(1, chord_progression * 2, time)  # 重复两次
    add_walking_bass(time)
    add_drum_fill(time + 15)  # 第15小节加花


# =====================
# 副歌部分 (8小节)
# =====================
def add_chorus():
    time = 24

    # 高潮旋律
    chorus_melody = [
        (76, 0, 0.75), (76, 0.75, 0.25), (74, 1, 0.5),  # 跳跃节奏
        (72, 1.5, 0.5), (74, 2, 2),  # 长音保持
        (72, 4, 1), (74, 5, 0.5), (76, 5.5, 2.5)  # 上升推进
    ]

    # 强力和弦
    power_chords = [
        ([60, 67], 0, 1), ([65, 72], 1, 1),  # 五度和声
        ([58, 65], 2, 1), ([67, 74], 3, 1),
        ([60, 67], 4, 2), ([72, 79], 6, 2)
    ]

    # 添加声部
    add_notes(0, chorus_melody, time)
    add_chords(1, power_chords, time)
    add_octave_bass(time)
    add_drums(3, [(36, 0, 0.25), (42, 0.25, 0.25)] * 8, time)  # 密集鼓点


# ====== 工具函数 ======
def add_notes(track, notes, start_time):
    """添加音符到指定轨道"""
    for note in notes:
        pitch = note[0]
        offset = note[1]
        duration = note[2]
        midi.addNote(track, 0, pitch, start_time + offset, duration, 100)


def add_chords(track, chords, start_time):
    """添加和弦"""
    for chord in chords:
        pitches, offset, duration = chord
        for pitch in pitches:
            midi.addNote(track, 0, pitch, start_time + offset, duration, 80)


def add_drums(track, pattern, start_time, repeat=1):
    """添加鼓节奏"""
    for i in range(repeat):
        for hit in pattern:
            midi.addNote(track, 9, hit[0], start_time + hit[1] + i * 4, hit[2], 100)


# ==== 新增的贝斯/鼓辅助函数 ====
def add_walking_bass(start_time):
    """Walking bass线"""
    for i in range(16):  # 16个四分音符
        note = 36 + (i % 4)
        midi.addNote(2, 0, note, start_time + i, 0.5, 90)

def add_drum_fill(fill_time):
    """鼓组加花"""
    drum_fill = [
        (38, 0, 0.25), (38, 0.5, 0.25),
        (42, 1, 0.25), (42, 1.5, 0.25)
    ]
    add_drums(3, drum_fill, fill_time)

def add_octave_bass(start_time):
    """八度贝斯"""
    for t in [0, 2, 4, 6]:
        midi.addNote(2, 0, 36, start_time + t, 2, 110)
        midi.addNote(2, 0, 48, start_time + t, 2, 90)


# ====== 生成段落 ======
add_intro()
add_verse()
add_chorus()

# 导出MIDI文件
with open("pop_song.mid", "wb") as f:
    midi.writeFile(f)
print("MIDI文件生成完成！")
```

```
from midiutil import MIDIFile

# 创建MIDI文件（2个音轨：0-吉他，1-鼓）
midi = MIDIFile(2)

# 基本设置
bpm = 160  # 朋克常用快速节奏
track = 0
time = 0  # 时间轴（单位：拍）
duration = 1  # 基础音符时长

# 吉他音轨设置（通道0）
midi.addTrackName(track, time, "Distortion Guitar")
midi.addProgramChange(track, 0, time, 30)  # 30:失真吉他音色
midi.addTempo(track, time, bpm)

# 鼓音轨设置（通道9，MIDI规范中通道10对应打击乐）
drum_track = 1
midi.addTrackName(drum_track, time, "Drum Kit")
midi.addProgramChange(drum_track, 9, time, 0)  # 标准鼓组

# 朋克常用强力和弦进行（根音+五度）
chord_progression = [
    # (根音MIDI编号, 五音MIDI编号)
    (28, 35),  # E5 (E2 + B2)
    (25, 32),  # A5 (A2 + E2)
    (23, 30),  # G5 (G2 + D3)
    (28, 35)  # E5
]

# 添加吉他强力和弦（每个和弦1小节）
for i in range(8):  # 重复8个小节
    for root, fifth in chord_progression:
        # 根音（力度加强）
        midi.addNote(track, 0, root, time, duration * 2, 100)
        # 五音
        midi.addNote(track, 0, fifth, time, duration * 2, 95)
        time += 2  # 每个和弦持续2拍

    time = 0 if (i + 1) % 4 == 0 else time  # 每4小节重置时间（循环结构）

# 添加鼓节奏（经典朋克节奏）
time = 0
drum_duration = 0.5  # 八分音符节奏

# 鼓音符映射（MIDI打击乐编号）
KICK = 36  # 底鼓
SNARE = 38  # 军鼓
HH_CLOSED = 42  # 闭合踩镲
CRASH = 49  # 吊镲

# 经典朋克鼓模式（2小节循环）
for _ in range(8):  # 重复4次（共8小节）
    for beat in range(8):  # 每小节8个八分音符
        current_time = time + beat * drum_duration

        # 底鼓（强拍）
        if beat % 4 in [0, 2]:  # 1、3拍
            midi.addNote(drum_track, 9, KICK, current_time, drum_duration, 90)

        # 军鼓（反拍）
        if beat % 4 in [1, 3]:  # 2、4拍
            midi.addNote(drum_track, 9, SNARE, current_time, drum_duration, 85)

        # 踩镲（持续八分音符）
        midi.addNote(drum_track, 9, HH_CLOSED, current_time, drum_duration, 70)

    # 每小节结束加吊镲
    midi.addNote(drum_track, 9, CRASH, time + 7 * drum_duration, drum_duration * 2, 100)
    time += 4  # 每小节4拍

# 保存文件
with open("punk_song.mid", "wb") as f:
    midi.writeFile(f)

print("MIDI文件生成完成！")
```

```
import pretty_midi
import math

# 创建MIDI对象
midi = pretty_midi.PrettyMIDI()

beat_duration = 60 / bpm  # 四分音符时长（秒）

# ======================
# 创建鼓轨（摇滚节奏）
# ======================
drum = pretty_midi.Instrument(0, is_drum=True)  # 通道9是鼓（program=0）

# 底鼓（每小节第1、3拍）
for i in range(8):
    start = i * 2 * beat_duration  # 每小节两拍（四四拍）
    drum.notes.append(pretty_midi.Note(
        velocity=100, pitch=36,  # 36=底鼓
        start=start, end=start + 0.1))

# 军鼓（每小节第2、4拍）
for i in range(8):
    start = (i * 2 + 1) * beat_duration
    drum.notes.append(pretty_midi.Note(
        velocity=90, pitch=38,  # 38=军鼓
        start=start, end=start + 0.1))

# 踩镲（八分音符）
for i in range(16):
    start = i * beat_duration / 2
    drum.notes.append(pretty_midi.Note(
        velocity=70, pitch=42,  # 42=闭镲
        start=start, end=start + 0.05))

# ======================
# 创建贝斯轨（根音驱动）
# ======================
bass = pretty_midi.Instrument(33)  # 电贝司
chord_progression = ['G2', 'C2', 'D2', 'G2']  # 和弦进行

for idx, chord in enumerate(chord_progression * 2):  # 重复两次
    note_number = pretty_midi.note_name_to_number(chord)
    start = idx * 4 * beat_duration  # 每和弦4拍
    bass.notes.append(pretty_midi.Note(
        velocity=80, pitch=note_number,
        start=start, end=start + 4 * beat_duration - 0.1))

# ======================
# 创建节奏吉他（强力和弦）
# ======================
guitar = pretty_midi.Instrument(30)  # 失真吉他
power_chords = {
    'G2': [43, 47, 50],  # G5和弦
    'C2': [48, 52, 55],  # C5和弦
    'D2': [50, 54, 57]   # D5和弦
}

for idx, chord in enumerate(chord_progression * 2):
    start = idx * 4 * beat_duration
    for pitch in power_chords[chord]:
        guitar.notes.append(pretty_midi.Note(
            velocity=70, pitch=pitch,
            start=start, end=start + 4 * beat_duration - 0.1,
        ))

# ======================
# 主旋律吉他（G大调旋律）
# ======================
lead = pretty_midi.Instrument(81)  # 合成领奏
melody_notes = [
    (67, 0.5), (65, 0.5), (67, 0.5), (65, 0.5),  # E-D-E-D
    (63, 1.0), (62, 1.0), (60, 2.0),            # C-B-G
    (62, 0.5), (63, 0.5), (65, 1.0), (63, 1.0)  # B-C-D-C
]

current_time = 8 * beat_duration  # 前8小节伴奏后进入
for pitch, duration in melody_notes * 2:
    lead.notes.append(pretty_midi.Note(
        velocity=90, pitch=pitch,
        start=current_time,
        end=current_time + duration * beat_duration
    ))
    current_time += duration * beat_duration

# 添加所有音轨
midi.instruments.extend([drum, bass, guitar, lead])

# 保存文件
midi.write('rock_song.mid')
print("MIDI文件已生成：rock_song.mid")
```

```
from pretty_midi import PrettyMIDI, Instrument, Note
import os


def create_gu_feng_midi(output_path="gu_feng_melody.mid"):
    """创建并保存古风MIDI文件"""

    # 1. 创建MIDI对象和古筝乐器
    midi = PrettyMIDI()
    gu_zheng = Instrument(program=24, is_drum=False, name="Gu Zheng")

    # 2. 定义五声音阶 (D调: 宫=D, 商=E, 角=F#, 徵=A, 羽=B)
    scale = {
        '宫': 50,  # D3
        '商': 52,  # E3
        '角': 54,  # F#3
        '徵': 57,  # A3
        '羽': 59  # B3
    }

    # 3. 明确定义每个音符 (开始时间, 持续时间, 音高, 力度)
    notes_data = [
        # 第一乐句
        (0.0, 1.0, scale['宫'], 80, "宫"),
        (1.0, 1.0, scale['商'], 70, "商"),
        (2.0, 1.0, scale['角'], 80, "角"),
        (3.0, 1.0, scale['羽'], 90, "羽"),

        # 第二乐句
        (4.0, 1.0, scale['徵'], 85, "徵"),
        (5.0, 0.5, scale['角'], 70, "角"),
        (5.5, 0.5, scale['商'], 70, "商"),
        (6.0, 2.0, scale['宫'], 90, "宫-长音"),

        # 高潮部分
        (8.0, 0.5, scale['羽'] + 12, 80, "高音羽"),
        (8.5, 0.5, scale['徵'] + 12, 70, "高音徵"),
        (9.0, 0.5, scale['商'] + 12, 80, "高音商"),
        (9.5, 0.5, scale['宫'] + 12, 90, "高音宫"),

        # 结尾
        (10.0, 1.0, scale['羽'], 85, "羽"),
        (11.0, 0.5, scale['徵'], 70, "徵"),
        (11.5, 0.5, scale['角'], 70, "角"),
        (12.0, 4.0, scale['宫'], 100, "宫-结束")
    ]

    # 4. 添加装饰音 (颤音效果)
    decorations = [
        (2.8, 0.2, scale['角'], 60, "滑音"),
        (3.8, 0.2, scale['羽'], 60, "滑音"),
        (5.8, 0.2, scale['商'], 60, "滑音"),
        (11.8, 0.2, scale['宫'], 60, "滑音")
    ]

    # 5. 创建并添加所有音符
    for start, duration, pitch, velocity, name in notes_data + decorations:
        note = Note(
            velocity=velocity,
            pitch=pitch,
            start=start,
            end=start + duration
        )
        gu_zheng.notes.append(note)
        print(f"添加音符: 时间{start:.1f}-{start + duration:.1f}, 音高{pitch}({name}), 力度{velocity}")

    # 6. 添加乐器并保存文件
    midi.instruments.append(gu_zheng)
    midi.write(output_path)

    # 7. 返回完整路径
    abs_path = os.path.abspath(output_path)
    print(f"\n古风MIDI已保存到: {abs_path}")
    return abs_path
```

### 古风

```
import mido
from mido import Message, MidiFile, MidiTrack
import random

# 中国五声音阶（C 宫调式）
chinese_pentatonic_scale = [60, 62, 64, 67, 69]

# 古典音乐节奏模式
classical_rhythm_patterns = [
    [1, 1, 1, 1],
    [2, 2],
    [1, 2, 1],
    [2, 1, 1],
    [0.5, 0.5, 0.5, 1.5],  # 附点节奏
    [0.33, 0.33, 0.33, 1, 1]  # 包含三连音
]

# 中国古典音乐常见和弦进行
chinese_chord_progression = [
    [60, 64, 67],
    [62, 67, 69],
    [64, 69, 72]
]

# 放慢节奏，每拍时长（毫秒）
beat_duration = 350
# 最大时长（一分钟）
max_duration = 60 * 1000

# 创建 MIDI 文件
mid = MidiFile()

# 主音轨，钢琴和古筝交替发声
main_track = MidiTrack()
mid.tracks.append(main_track)

# 鼓点音轨
drum_track = MidiTrack()
mid.tracks.append(drum_track)
# 设置鼓组通道
drum_track.append(Message('program_change', program=0, channel=9, time=0))

# 更慢、更具古风规律的鼓点节奏模式
drum_rhythm_patterns = [
    [2, 0, 2, 0],  # 每两拍一个鼓点，且间隔两拍
    [2, 0, 0, 2],
    [1, 0, 1, 0, 1, 0, 1, 0],  # 较慢的均匀节奏
    [3, 0, 1, 0]  # 有长音和短音的组合
]

# 更具古风厚重感的鼓点音色
drum_notes = [35, 36, 43]  # 35: 大鼓, 36: 低音鼓, 43: 落地鼓

# 原有的吹箫声背景旋律和声音轨
flute_track_1 = MidiTrack()
mid.tracks.append(flute_track_1)
# 设置吹箫乐器
flute_track_1.append(Message('program_change', program=73, time=0))

# 新增的吹箫音轨
flute_track_2 = MidiTrack()
mid.tracks.append(flute_track_2)
# 设置吹箫乐器
flute_track_2.append(Message('program_change', program=73, time=0))

current_time = 0
chord_index = 0
prev_note = None
# 用于交替选择钢琴和古筝
instrument_turn = 0

while current_time < max_duration:
    # 选择当前和弦
    current_chord = chinese_chord_progression[chord_index]
    # 选择一个主音节奏模式
    main_rhythm = random.choice(classical_rhythm_patterns)
    # 选择一个鼓点节奏模式
    drum_rhythm = random.choice(drum_rhythm_patterns)
    # 为新增吹箫音轨选择一个节奏模式
    flute_2_rhythm = random.choice(classical_rhythm_patterns)

    main_rhythm_index = 0
    drum_rhythm_index = 0
    flute_2_rhythm_index = 0

    while main_rhythm_index < len(main_rhythm):
        if current_time >= max_duration:
            break

        main_note_length = main_rhythm[main_rhythm_index]
        # 计算音符持续时间对应的 MIDI 时间
        midi_time = mido.second2tick(main_note_length * beat_duration / 1000, mid.ticks_per_beat, mido.bpm2tempo(120))

        # 选择音符，控制旋律走向
        if prev_note is None:
            note = random.choice(current_chord)
        else:
            possible_notes = []
            for candidate_note in chinese_pentatonic_scale:
                interval = abs(candidate_note - prev_note)
                if interval <= 3:
                    possible_notes.append(candidate_note)
            if not possible_notes:
                possible_notes = chinese_pentatonic_scale
            note = random.choice(possible_notes)
        prev_note = note

        # 交替选择钢琴和古筝
        if instrument_turn % 2 == 0:
            program = 0  # 钢琴
            velocity = random.randint(70, 90)
        else:
            program = 46  # 古筝
            velocity = random.randint(60, 80)

        # 发送乐器变更消息
        if instrument_turn % 2 == 0:
            main_track.append(Message('program_change', program=0, time=0))
        else:
            main_track.append(Message('program_change', program=46, time=0))

        # 发送音符开启和关闭消息
        main_track.append(Message('note_on', note=note, velocity=velocity, time=0))
        main_track.append(Message('note_off', note=note, velocity=velocity, time=int(midi_time)))

        # 添加鼓点
        if drum_rhythm_index < len(drum_rhythm) and drum_rhythm[drum_rhythm_index] > 0:
            drum_note = random.choice(drum_notes)
            drum_velocity = random.randint(80, 100)  # 增加力度，让声音更厚重
            drum_midi_time = mido.second2tick(drum_rhythm[drum_rhythm_index] * beat_duration / 1000, mid.ticks_per_beat,
                                              mido.bpm2tempo(120))
            drum_track.append(Message('note_on', note=drum_note, channel=9, velocity=drum_velocity, time=0))
            drum_track.append(
                Message('note_off', note=drum_note, channel=9, velocity=drum_velocity, time=int(drum_midi_time)))

        # 添加原有的吹箫声背景旋律和声
        flute_1_note = random.choice(current_chord)
        flute_1_velocity = random.randint(40, 60)  # 吹箫声相对轻柔
        flute_track_1.append(Message('note_on', note=flute_1_note, velocity=flute_1_velocity, time=0))
        flute_track_1.append(Message('note_off', note=flute_1_note, velocity=flute_1_velocity, time=int(midi_time)))

        # 添加新增的吹箫声音
        if flute_2_rhythm_index < len(flute_2_rhythm):
            flute_2_note_length = flute_2_rhythm[flute_2_rhythm_index]
            flute_2_midi_time = mido.second2tick(flute_2_note_length * beat_duration / 1000, mid.ticks_per_beat,
                                                 mido.bpm2tempo(120))
            flute_2_note = random.choice(current_chord)
            flute_2_velocity = random.randint(40, 60)  # 吹箫声相对轻柔
            flute_track_2.append(Message('note_on', note=flute_2_note, velocity=flute_2_velocity, time=0))
            flute_track_2.append(
                Message('note_off', note=flute_2_note, velocity=flute_2_velocity, time=int(flute_2_midi_time)))
            flute_2_rhythm_index += 1

        main_rhythm_index += 1
        drum_rhythm_index = (drum_rhythm_index + 1) % len(drum_rhythm)
        instrument_turn += 1
        current_time += main_note_length * beat_duration

    # 切换到下一个和弦
    chord_index = (chord_index + 1) % len(chinese_chord_progression)

# 保存 MIDI 文件
mid.save('222.mid')
```
