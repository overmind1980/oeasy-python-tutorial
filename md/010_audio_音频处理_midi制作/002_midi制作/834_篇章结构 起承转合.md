### 起承转合
 
 ![图片描述](https://doc.shiyanlou.com/courses/3584/labs/4071390/uid1190679-20250409-1744159290483) 

```
import mido
from mido import MidiFile, MidiTrack, Message, MetaMessage

# 创建MIDI文件和轨道
mid = MidiFile(ticks_per_beat=480)  # 设置时间精度为480 ticks/beat
melody_track = MidiTrack()
mid.tracks.append(melody_track)

# 设置乐器为钢琴
melody_track.append(Message('program_change', program=0, time=0))  # 钢琴

# 设置拍号为3/4
time_signature = MetaMessage('time_signature', numerator=3, denominator=4, 
                            clocks_per_click=24, notated_32nd_notes_per_beat=8, time=0)
melody_track.append(time_signature)

# 生日歌主旋律 (3/4拍)
birthday_melody = [
        # 第一乐句 (2个小节): "祝你生日快乐"
        (60, 0.5), (60, 0.5), (62, 1), (60, 1), (65, 1), (64, 2),
        
        # 第二乐句 (2个小节): "祝你生日快乐"
        (60, 0.5), (60, 0.5), (62, 1), (60, 1), (67, 1), (65, 2),
        
        # 第三乐句 (2个小节+额外1拍): "祝你生日快乐，亲爱的xxx"
        (60, 0.5), (60, 0.5), (72, 1), (69, 1), (65, 1), (64, 1), (62, 2),
        
        # 第四乐句 (2个小节): "祝你生日快乐"
        (70, 0.5), (70, 0.5), (69, 1), (65, 1), (67, 1), (65, 2)
]

# 添加音符到轨道
current_time = 0
for note, duration in birthday_melody:
    # 音符开始
    melody_track.append(Message('note_on', note=note, velocity=64, time=current_time if current_time == 0 else 0))
    
    # 音符结束
    ticks = int(duration * 480)  # 将时长转换为ticks
    melody_track.append(Message('note_off', note=note, velocity=64, time=ticks))
    
    current_time = 0  # 重置时间，因为我们使用相对时间

# 保存MIDI文件
mid.save('simple_birthday.mid')

print("简单的生日歌MIDI文件已成功保存为 'simple_birthday.mid'")

```

def create_midi(filename, melody_data, rhythm_data):
    """从旋律数据和节奏数据创建MIDI文件"""
    with open(filename, 'wb') as f:
        # 写入文件头
        f.write(b'MThd')
        f.write(struct.pack('>I', 6))  # 头长度
        # 格式1，音轨数，时间分隔符
        num_tracks = 2  # 旋律轨 + 节奏轨
        f.write(struct.pack('>HHH', 1, num_tracks, 480))
        
        # 处理旋律轨道
        track_buffer = bytearray()
        
        # 写入音轨名
        track_buffer.extend(b'\x00\xFF\x03')
        track_name = b'Melody Track'
        # 写入可变长度整数
        val = len(track_name)
        if val < 0x80:
            track_buffer.append(val)
        elif val < 0x4000:
            track_buffer.append((val >> 7) | 0x80)
            track_buffer.append(val & 0x7F)
        elif val < 0x200000:
            track_buffer.append((val >> 14) | 0x80)
            track_buffer.append(((val >> 7) & 0x7F) | 0x80)
            track_buffer.append(val & 0x7F)
        else:
            track_buffer.append((val >> 21) | 0x80)
            track_buffer.append(((val >> 14) & 0x7F) | 0x80)
            track_buffer.append(((val >> 7) & 0x7F) | 0x80)
            track_buffer.append(val & 0x7F)
        track_buffer.extend(track_name)
        
        # 处理音符事件
        last_time = 0
        for note in melody_data[0]:
            start, duration, pitch, velocity = note
            delta_time = start - last_time
            
            # 写入delta time
            val = delta_time
            if val < 0x80:
                track_buffer.append(val)
            elif val < 0x4000:
                track_buffer.append((val >> 7) | 0x80)
                track_buffer.append(val & 0x7F)
            elif val < 0x200000:
                track_buffer.append((val >> 14) | 0x80)
                track_buffer.append(((val >> 7) & 0x7F) | 0x80)
                track_buffer.append(val & 0x7F)
            else:
                track_buffer.append((val >> 21) | 0x80)
                track_buffer.append(((val >> 14) & 0x7F) | 0x80)
                track_buffer.append(((val >> 7) & 0x7F) | 0x80)
                track_buffer.append(val & 0x7F)
            
            # 写入音符开始事件
            track_buffer.append(0x90)  # note_on, channel 0
            track_buffer.append(pitch)
            track_buffer.append(velocity)
            
            # 写入delta time for note off
            val = duration
            if val < 0x80:
                track_buffer.append(val)
            elif val < 0x4000:
                track_buffer.append((val >> 7) | 0x80)
                track_buffer.append(val & 0x7F)
            elif val < 0x200000:
                track_buffer.append((val >> 14) | 0x80)
                track_buffer.append(((val >> 7) & 0x7F) | 0x80)
                track_buffer.append(val & 0x7F)
            else:
                track_buffer.append((val >> 21) | 0x80)
                track_buffer.append(((val >> 14) & 0x7F) | 0x80)
                track_buffer.append(((val >> 7) & 0x7F) | 0x80)
                track_buffer.append(val & 0x7F)
            
            # 写入音符结束事件
            track_buffer.append(0x80)  # note_off, channel 0
            track_buffer.append(pitch)
            track_buffer.append(0)
            
            last_time = start + duration
        
        # 添加音轨结束事件
        track_buffer.extend(b'\x00\xFF\x2F\x00')
        
        # 写入音轨头
        f.write(b'MTrk')
        track_length = len(track_buffer)
        f.write(struct.pack('>I', track_length))
        
        # 写入音轨数据
        f.write(track_buffer)
        
        # 处理打击乐轨道
        track_buffer = bytearray()
        
        # 写入音轨名
        track_buffer.extend(b'\x00\xFF\x03')
        track_name = b'Rhythm Track'
        # 写入可变长度整数
        val = len(track_name)
        if val < 0x80:
            track_buffer.append(val)
        elif val < 0x4000:
            track_buffer.append((val >> 7) | 0x80)
            track_buffer.append(val & 0x7F)
        elif val < 0x200000:
            track_buffer.append((val >> 14) | 0x80)
            track_buffer.append(((val >> 7) & 0x7F) | 0x80)
            track_buffer.append(val & 0x7F)
        else:
            track_buffer.append((val >> 21) | 0x80)
            track_buffer.append(((val >> 14) & 0x7F) | 0x80)
            track_buffer.append(((val >> 7) & 0x7F) | 0x80)
            track_buffer.append(val & 0x7F)
        track_buffer.extend(track_name)
        
        # 处理打击乐事件
        last_time = 0
        for note in rhythm_data:
            start, duration, pitch, velocity = note
            delta_time = start - last_time
            
            # 写入delta time
            val = delta_time
            if val < 0x80:
                track_buffer.append(val)
            elif val < 0x4000:
                track_buffer.append((val >> 7) | 0x80)
                track_buffer.append(val & 0x7F)
            elif val < 0x200000:
                track_buffer.append((val >> 14) | 0x80)
                track_buffer.append(((val >> 7) & 0x7F) | 0x80)
                track_buffer.append(val & 0x7F)
            else:
                track_buffer.append((val >> 21) | 0x80)
                track_buffer.append(((val >> 14) & 0x7F) | 0x80)
                track_buffer.append(((val >> 7) & 0x7F) | 0x80)
                track_buffer.append(val & 0x7F)
            
            # 写入音符开始事件（使用通道9，即10通道，从1开始计数）
            track_buffer.append(0x99)  # note_on, channel 9 (打击乐通道)
            track_buffer.append(pitch)
            track_buffer.append(velocity)
            
            # 写入delta time for note off
            val = duration
            if val < 0x80:
                track_buffer.append(val)
            elif val < 0x4000:
                track_buffer.append((val >> 7) | 0x80)
                track_buffer.append(val & 0x7F)

                track_buffer.append(val & 0x7F)

            track_buffer.append(0x89)  # note_off, channel 9
            track_buffer.append(pitch)
            track_buffer.append(0)

META_TYPES = {
    "end_of_track",
    "key_signature",
    "lyrics",
    "set_tempo",
    "text",
    "time_signature",
    "track_name",
}


def _make_msg(track_name, d):
    msg_type = d.get("type")
    time = int(d.get("time", 0) or 0)

    if msg_type in META_TYPES:
        kwargs = {k: v for k, v in d.items() if k != "type"}
        kwargs["time"] = time

        if msg_type == "set_tempo":
            kwargs.setdefault("tempo", 500000)
        elif msg_type == "time_signature":
            kwargs.setdefault("numerator", 4)
            kwargs.setdefault("denominator", 4)
            kwargs.setdefault("clocks_per_click", 24)
            kwargs.setdefault("notated_32nd_notes_per_beat", 8)
        elif msg_type == "track_name":
            kwargs.setdefault("name", track_name or "")
        elif msg_type == "key_signature":
            kwargs.setdefault("key", "C")
        elif msg_type in ("lyrics", "text"):
            kwargs.setdefault("text", "")

        return mido.MetaMessage(msg_type, **kwargs)

    kwargs = {k: v for k, v in d.items() if k != "type"}
    kwargs["time"] = time

    if msg_type == "pitchwheel":
        kwargs.setdefault("pitch", 0)
        kwargs.setdefault("channel", 0)

    if msg_type in ("note_on", "note_off", "control_change", "program_change"):
        kwargs.setdefault("channel", 0)

    return mido.Message(msg_type, **kwargs)


def build_midi():
    mid = mido.MidiFile(type=int(DATA.get("format", 1)), ticks_per_beat=int(DATA.get("ticks_per_beat", 96)))

    for tr in DATA.get("tracks", []):
        name = tr.get("name", "")
        track = mido.MidiTrack()
        for d in tr.get("messages", []):
            track.append(_make_msg(name, d))
        mid.tracks.append(track)

    return mid


def main():
    default_out = os.path.splitext(os.path.basename(__file__))[0] + ".mid"
    out_path = sys.argv[1] if len(sys.argv) > 1 else default_out
    build_midi().save(out_path)
    print(out_path)


if __name__ == "__main__":
