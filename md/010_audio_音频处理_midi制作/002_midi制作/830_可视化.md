import pretty_midi
import numpy as np
import matplotlib.pyplot as plt

def visualize_midi(midi_file):
    # 加载MIDI文件
    pm = pretty_midi.PrettyMIDI(midi_file)
    
    # 创建图形
    plt.figure(figsize=(15, 8))
    
    # 为每个乐器创建子图
    for i, instrument in enumerate(pm.instruments):
        # 获取音符
        notes = instrument.notes
        
        # 绘制音符
        for note in notes:
            # 创建矩形
            rect = plt.Rectangle(
                (note.start, note.pitch),
                note.end - note.start,
                0.5,
                alpha=0.5,
                color='blue' if not instrument.is_drum else 'red'
            )
            plt.gca().add_patch(rect)
    
    # 设置坐标轴
    plt.ylim(20, 100)  # MIDI音符范围
    plt.xlim(0, pm.get_end_time())
    
    # 添加网格
    plt.grid(True)
    
    # 添加标签
    plt.xlabel('时间 (秒)')
    plt.ylabel('音高 (MIDI音符号)')
    plt.title('MIDI可视化')
    
    # 显示图形
    plt.show()

# 可视化生成的MIDI文件
visualize_midi('mashiro_new.mid')
