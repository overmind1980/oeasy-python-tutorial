```
from mido import Message, MidiFile, MidiTrack
# 修改导入路径
from mido import MetaMessage

# 创建一个新的MIDI文件
mid = MidiFile()
# 创建一个新的音轨
track = MidiTrack()
mid.tracks.append(track)

# 设置时间签名（4/4拍）
time_signature_msg = MetaMessage('time_signature', numerator=4, denominator=2, clocks_per_click=24, notated_32nd_notes_per_beat=8)
track.append(time_signature_msg)

# 设置速度（这里设置为120 BPM，转换为微秒/拍）
tempo = 500000  # 120 BPM对应的微秒/拍
tempo_msg = MetaMessage('set_tempo', tempo=tempo)
track.append(tempo_msg)

# 定义一些音符和演奏法设置（这里简单模拟连音和断奏）
# 连音：音符之间没有间隔
# 断奏：音符之间有较短的间隔

# 连音部分
notes = [60, 62, 64]  # 音符的MIDI编号（C4, D4, E4）
for note in notes:
    track.append(Message('note_on', note=note, velocity=64, time=0))
    track.append(Message('note_off', note=note, velocity=64, time=480))  # 音符持续时间

# 断奏部分
staccato_notes = [65, 67, 69]  # 音符的MIDI编号（F4, G4, A4）
for staccato_note in staccato_notes:
    track.append(Message('note_on', note=staccato_note, velocity=64, time=0))
    track.append(Message('note_off', note=staccato_note, velocity=64, time=240))  # 较短的持续时间，模拟断奏
    track.append(Message('note_on', note=staccato_note, velocity=0, time=240))  # 间隔时间

# 保存MIDI文件
mid.save('demo_midi_with_articulation.mid')
```

### 滑音效果

```
from mido import Message, MidiFile, MidiTrack
from mido import MetaMessage  # 导入 MetaMessage

mid = MidiFile()
track = MidiTrack()
mid.tracks.append(track)

# 设置速度为120 BPM
tempo = 500000  
track.append(MetaMessage('set_tempo', tempo=tempo))  # 使用 MetaMessage

# 起始音符和目标音符
start_note = 60  # C4
end_note = 64  # E4

# 发送起始音符开启消息
track.append(Message('note_on', note=start_note, velocity=64, time=0))

# 逐渐改变音高实现滑音
pitch_bend_range = 2  # 音高弯曲范围（半音数）
pitch_bend_steps = 10  # 分多少步完成滑音
for step in range(pitch_bend_steps):
    # 计算当前音高弯曲值，并确保在 -8192 到 8191 之间
    pitch_bend_value = int((end_note - start_note) * step / pitch_bend_steps * 8191 / pitch_bend_range)
    pitch_bend_value = max(-8192, min(8191, pitch_bend_value))  # 限制值的范围
    track.append(Message('pitchwheel', pitch=pitch_bend_value, time=50))  # 使用 pitchwheel

# 发送目标音符的保持消息（可以根据需要调整持续时间）
track.append(Message('note_on', note=end_note, velocity=64, time=200))
track.append(Message('note_off', note=end_note, velocity=64, time=200))

mid.save('glissando.mid')
```

### 笛子 装饰音

```
import mido
from mido import Message, MidiFile, MidiTrack, MetaMessage

# 初始化MIDI文件
mid = MidiFile(ticks_per_beat=480)
tempo = mido.bpm2tempo(68)  # 整体速度控制

# 创建三个独立音轨
flute_track = MidiTrack()   # 笛子音轨（通道0）
guzheng_track = MidiTrack() # 古筝音轨（通道1）
piano_track = MidiTrack()   # 钢琴音轨（通道2）
mid.tracks.extend([flute_track, guzheng_track, piano_track])

# 乐器初始化
flute_track.append(Message('program_change', program=75, time=0))   # 笛子
guzheng_track.append(Message('program_change', program=107, time=0, channel=1)) # 古筝
piano_track.append(Message('program_change', program=0, time=0, channel=2))     # 钢琴

# ===== 悲伤段落（笛子独奏）=====
def add_flute_intro(track):
    """婉转的笛子前奏，使用装饰音技法"""
    notes = [
        (76, 1.5, 60), (74, 0.5, 70), (72, 2, 50),  # 装饰音处理
        (71, 1, 60), (69, 0.5, 50), (67, 3, 40),    # 渐弱处理
        (0, 1)  # 休止符
    ]
    for note in notes:
        if note[0] == 0:  # 处理休止符
            track.append(Message('note_off', note=note[0], velocity=0, time=int(480*note[1])))
            continue
        track.append(Message('note_on', note=note[0], velocity=note[2], time=0))
        track.append(Message('note_off', note=note[0], velocity=note[2], time=int(480*note[1])))
    # 添加颤音效果（MIDI控制器#1）
    track.append(Message('control_change', control=1, value=90, time=0))

add_flute_intro(flute_track)

# ===== 过渡段（加入古筝泛音）=====
guzheng_track.append(Message('control_change', control=11, value=60, time=480)) # 表情控制
guzheng_glissando = [
    (84, 0.5, 30), (83, 0.5, 40), (81, 0.5, 50), (79, 1.5, 60)  # 刮奏效果
]
for note in guzheng_glissando:
    guzheng_track.append(Message('note_on', note=note[0], velocity=note[2], time=0))
    guzheng_track.append(Message('note_off', note=note[0], velocity=note[2], time=int(480*note[1])))

# ===== 激昂段落（古筝与钢琴对话）=====
def heroic_section(guzheng, piano):
    # 古筝扫弦节奏
    guzheng_chords = [
        ([67,71,74], 1, 80), ([65,69,72], 0.5, 70),
        ([64,67,72], 1, 90), ([62,65,69], 0.5, 60)
    ]
    
    # 钢琴分解和弦
    piano_arpeggio = [
        (60, 0.25, 50), (64, 0.25, 60), (67, 0.5, 70), 
        (62, 0.25, 50), (65, 0.25, 60), (69, 0.5, 70)
    ]
    
    # 交替演奏模式
    for i in range(4):
        # 古筝强力和弦
        for note in guzheng_chords[i%4][0]:
            guzheng.append(Message('note_on', note=note, velocity=guzheng_chords[i%4][2], time=0))
        # 钢琴分解和弦
        for idx, note in enumerate(piano_arpeggio):
            on_time = 0 if idx==0 else int(480*piano_arpeggio[idx-1][1])
            piano.append(Message('note_on', note=note[0], velocity=note[2], time=on_time))
            piano.append(Message('note_off', note=note[0], time=int(480*note[1])))
        # 关闭古筝和弦
        for note in guzheng_chords[i%4][0]:
            guzheng.append(Message('note_off', note=note, time=int(480*guzheng_chords[i%4][1])))

heroic_section(guzheng_track, piano_track)

# ===== 释然段落（乐器融合）=====
def resolution_section():
    # 古筝长音铺垫
    guzheng_track.append(Message('control_change', control=11, value=40, time=960)) # 减弱表情
    guzheng_track.append(Message('note_on', note=72, velocity=50, time=0))
    guzheng_track.append(Message('note_off', note=72, time=1920))
    
    # 钢琴流动音型
    piano_notes = [(60,0.5), (64,0.5), (67,1), (69,0.5), (67,0.5), (64,2)]
    for note in piano_notes:
        piano_track.append(Message('note_on', note=note[0], velocity=40, time=0))
        piano_track.append(Message('note_off', note=note[0], time=int(480*note[1])))
    
    # 笛子再现主题
    flute_track.append(Message('control_change', control=1, value=30, time=1440)) # 减少颤音
    flute_track.append(Message('note_on', note=69, velocity=50, time=0))
    flute_track.append(Message('note_off', note=69, time=960))

resolution_section()

# ===== 结尾处理 =====
# 渐慢处理（通过调整tempo）
mid.tracks[0].append(MetaMessage('set_tempo', tempo=mido.bpm2tempo(60), time=1920))

# 保存文件
mid.save('emotional_journey.mid')
```

```
from midiutil import MIDIFile
import time

def create_advanced_melody():
    # 创建一个MIDI文件，使用6个轨道（主旋律、和弦、打击乐等）
    midi = MIDIFile(6)
    
    # 设置基本参数
    track = 0
    channel = 0
    time = 0
    tempo = 95  # 稍微快一点的中等速度
    volume = 100
    
    # 设置时间签名
    midi.addTempo(track, time, tempo)
    
    # 定义音符（使用MIDI音符编号）
    # 主旋律 - 更加跳跃的旋律线
    melody_notes = [
        # 第一段 - 活泼的跳跃
        88, 95, 88, 98, 88, 100, 88, 102,  # 八度跳跃
        91, 98, 91, 100, 91, 102, 91, 105,  # 大跳进
        93, 100, 93, 103, 93, 105, 93, 107,  # 更大的跳跃
        95, 102, 95, 105, 95, 107, 95, 108,  # 最高音区跳跃
        
        # 第二段 - 活泼的装饰音
        93, 95, 91, 96, 90, 97, 88, 98,  # 上行跳跃
        98, 96, 97, 95, 96, 94, 95, 93,  # 下行跳跃
        
        # 第三段 - 跳跃的音阶
        88, 91, 88, 93, 88, 95, 88, 98,  # 基音跳跃
        98, 95, 98, 93, 98, 91, 98, 88,  # 高音跳跃
        
        # 第四段 - 华丽的结束
        95, 88, 98, 91, 100, 93, 102, 95,  # 交错上升
        102, 95, 100, 93, 98, 91, 95, 88   # 交错下降
    ]
    
    # 定义音符持续时间（以拍为单位）
    melody_durations = [
        # 第一段 - 跳跃的节奏
        0.5, 0.25, 0.5, 0.25, 0.5, 0.25, 0.5, 0.25,  # 切分音节奏
        0.5, 0.25, 0.5, 0.25, 0.5, 0.25, 0.5, 0.25,
        0.5, 0.25, 0.5, 0.25, 0.5, 0.25, 0.5, 0.25,
        0.5, 0.25, 0.5, 0.25, 0.5, 0.25, 0.5, 0.25,
        
        # 第二段 - 活泼的节奏
        0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25,  # 快速的装饰音
        0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25,
        
        # 第三段 - 跳跃的节奏
        0.5, 0.25, 0.5, 0.25, 0.5, 0.25, 0.5, 0.25,  # 切分音节奏
        0.5, 0.25, 0.5, 0.25, 0.5, 0.25, 0.5, 0.25,
        
        # 第四段 - 华丽的结束
        0.75, 0.25, 0.75, 0.25, 0.75, 0.25, 0.75, 0.25,  # 第一遍
        0.75, 0.25, 0.75, 0.25, 0.75, 0.25, 1.0, 1.5   # 结尾延长
    ]
    
    # 添加主旋律
    current_time = 0
    for i, (note, duration) in enumerate(zip(melody_notes, melody_durations)):
        # 计算渐变音量
        if i >= len(melody_notes) - 12:  # 最后12个音符开始渐变
            fade_factor = (len(melody_notes) - i) / 12
            current_volume = int(volume * fade_factor)
        else:
            # 在高音部分增加音量
            if note >= 102:
                current_volume = min(127, volume + 27)
            elif note >= 98:
                current_volume = min(120, volume + 20)
            else:
                current_volume = volume
            
        midi.addNote(track, channel, note, current_time, duration, current_volume)
        current_time += duration
    
    # 添加和弦伴奏（更活泼的和弦进行）
    chord_notes = [
        [48, 52, 55, 59],  # C大七和弦
        [50, 53, 57, 60],  # D小七和弦
        [52, 55, 59, 62],  # E小七和弦
        [53, 57, 60, 64],  # F大七和弦
        [55, 59, 62, 65],  # G七和弦
        [57, 60, 64, 67],  # A小七和弦
        [59, 62, 65, 69],  # B七和弦
        [60, 64, 67, 71],  # C大七和弦
        [62, 65, 69, 72],  # D七和弦
        [64, 67, 71, 74],  # E七和弦
        [65, 69, 72, 76],  # F七和弦
        [67, 71, 74, 77]   # G七和弦
    ]
    
    # 添加和弦（更活泼的节奏）
    chord_time = 0
    for i, chord in enumerate(chord_notes):
        if i >= len(chord_notes) - 3:
            fade_factor = (len(chord_notes) - i) / 3
            current_volume = int((volume - 20) * fade_factor)
        else:
            current_volume = volume - 20
            
        # 分解和弦
        for j, note in enumerate(chord):
            midi.addNote(1, channel, note, chord_time + j*0.25, 1.0, current_volume)
        chord_time += 2.0
    
    # 添加打击乐器（使用MIDI通道10，这是打击乐专用通道）
    percussion_channel = 9  # MIDI通道10（0-15，所以9代表通道10）
    
    # 定义打击乐器音符（MIDI打击乐器音符）
    percussion_notes = {
        'kick': 36,      # 低音鼓
        'snare': 38,     # 军鼓
        'hihat': 42,     # 高音镲
        'tom': 45,       # 中音鼓
        'crash': 49,     # 吊镲
        'ride': 51,      # 骑镲
        'clap': 39,      # 拍手声
        'tambourine': 54 # 铃鼓
    }
    
    # 添加打击乐节奏（更加活泼的节奏）
    percussion_pattern = [
        # 基本节奏
        ('kick', 0.5), ('hihat', 0.25), ('clap', 0.25), ('snare', 0.5),
        ('tambourine', 0.25), ('hihat', 0.25), ('kick', 0.5), ('snare', 0.5),
        ('kick', 0.5), ('hihat', 0.25), ('clap', 0.25), ('snare', 0.5),
        ('tambourine', 0.25), ('hihat', 0.25), ('kick', 0.5), ('snare', 0.5),
        
        # 变化节奏
        ('kick', 0.25), ('hihat', 0.25), ('clap', 0.25), ('snare', 0.25),
        ('kick', 0.25), ('hihat', 0.25), ('tambourine', 0.25), ('snare', 0.25),
        ('kick', 0.25), ('hihat', 0.25), ('clap', 0.25), ('snare', 0.25),
        ('kick', 0.25), ('hihat', 0.25), ('tambourine', 0.25), ('snare', 0.25),
        
        # 高潮部分
        ('kick', 0.25), ('hihat', 0.25), ('clap', 0.25), ('snare', 0.25),
        ('tom', 0.25), ('hihat', 0.25), ('tambourine', 0.25), ('snare', 0.25),
        ('kick', 0.25), ('hihat', 0.25), ('clap', 0.25), ('snare', 0.25),
        ('tom', 0.25), ('hihat', 0.25), ('tambourine', 0.25), ('crash', 0.25),
        
        # 结尾 - 使用更柔和的乐器组合
        ('tambourine', 0.5), ('hihat', 0.5), ('ride', 0.5), ('hihat', 0.5),  # 轻柔的过渡
        ('tambourine', 0.75), ('ride', 0.75), ('hihat', 0.5), ('tambourine', 1.0)  # 优雅的结束
    ]
    
    # 添加打击乐
    percussion_time = 0
    for instrument, duration in percussion_pattern:
        midi.addNote(2, percussion_channel, percussion_notes[instrument], 
                    percussion_time, duration, volume)
        percussion_time += duration
    
    # 保存MIDI文件
    with open("advanced_melody_jumpy_v2.mid", "wb") as output_file:
        midi.writeFile(output_file)
    
    print("高级MIDI文件已生成：advanced_melody_jumpy_v2.mid")
    print("\n音符序列（MIDI音符编号）：")
    for i, note in enumerate(melody_notes):
        print(f"音符 {i+1}: {note}")

if __name__ == "__main__":
    create_advanced_melody()
```
