"""
快节奏音乐生成器（2分钟版本）
依赖库：music21==7.3.7, mido==1.2.10
"""
import random
import mido
from mido import MidiFile, MidiTrack, Message
from music21 import scale, chord, pitch

# 参数配置
BPM = 140  # 加快节奏
TRACKS = {
    "melody": {"channel": 0, "program": 0, "octave": 4},  # 钢琴
    "chord": {"channel": 1, "program": 48, "octave": 3},  # 弦乐
    "bass": {"channel": 2, "program": 32, "octave": 2},   # 贝斯
    "drum": {"channel": 9, "program": 0}                  # 打击乐
}
KEY = 'C'
SCALE_TYPE = 'major'
TOTAL_BARS = 64  # 64小节 ≈ 2分钟(140BPM)

def safe_get_pitches(scale_obj, octave):
    """安全获取音高列表"""
    try:
        return [p.midi for p in scale_obj.getPitches(f'{KEY}{octave}', f'{KEY}{octave+1}')]
    except:
        return [p.midi for p in scale_obj.getPitches(octave*12, (octave+2)*12)]

def generate_melody_track(mid):
    """生成快速旋律"""
    track = MidiTrack()
    track.name = "melody"
    track.append(Message('program_change', program=TRACKS["melody"]["program"]))

    scale_obj = scale.MajorScale(KEY) if SCALE_TYPE == 'major' else scale.MinorScale(KEY)
    pitches = safe_get_pitches(scale_obj, TRACKS["melody"]["octave"])

    current_time = 0
    for i in range(TOTAL_BARS * 4):  # 每小节4拍
        # 随机选择节奏型：1/8或1/16音符
        dur = int(mid.ticks_per_beat * (0.5 if random.random() > 0.3 else 0.25))
        note = random.choice(pitches)

        # 增加旋律变化（每8小节升高八度）
        if i % 32 == 0 and i > 0:
            pitches = [p+12 if p < 100 else p for p in pitches]

        track.append(Message('note_on', note=note, velocity=90, time=current_time))
        track.append(Message('note_off', note=note, velocity=90, time=dur))
        current_time = 0

    track.append(mido.MetaMessage('end_of_track', time=0))
    mid.tracks.append(track)

def generate_chord_progression():
    """生成扩展的和弦进行（I-V-vi-IV）"""
    scale_obj = scale.MajorScale(KEY)
    return [
        chord.Chord([scale_obj.pitchFromDegree(1), scale_obj.pitchFromDegree(3), scale_obj.pitchFromDegree(5)]),  # I
        chord.Chord([scale_obj.pitchFromDegree(5), scale_obj.pitchFromDegree(7), scale_obj.pitchFromDegree(9)]),  # V
        chord.Chord([scale_obj.pitchFromDegree(6), scale_obj.pitchFromDegree(8), scale_obj.pitchFromDegree(10)]), # vi
        chord.Chord([scale_obj.pitchFromDegree(4), scale_obj.pitchFromDegree(6), scale_obj.pitchFromDegree(8)])   # IV
    ]

def generate_chord_track(mid, chord_prog):
    """快速分解和弦"""
    track = MidiTrack()
    track.name = "chord"
    track.append(Message('program_change', program=TRACKS["chord"]["program"]))

    current_time = 0
    for _ in range(TOTAL_BARS // 4):  # 每个和弦4小节
        for chord_obj in chord_prog:
            notes = [p.midi for p in chord_obj.pitches]

            # 快速琶音模式
            for i in range(8):  # 每小节8个1/8音符
                note = notes[i % len(notes)]
                track.append(Message('note_on', note=note, velocity=75, time=current_time))
                track.append(Message('note_off', note=note, velocity=75, time=int(mid.ticks_per_beat/2)))
                current_time = 0

    track.append(mido.MetaMessage('end_of_track', time=0))
    mid.tracks.append(track)

def generate_bass_track(mid, chord_prog):
    """驱动型贝斯"""
    track = MidiTrack()
    track.name = "bass"
    track.append(Message('program_change', program=TRACKS["bass"]["program"]))

    current_time = 0
    for _ in range(TOTAL_BARS // 4):
        for chord_obj in chord_prog:
            root = chord_obj.root().midi - 12
            # 八度跳跃节奏
            for i in range(4):  # 每小节4个1/4音符
                note = root if i % 2 == 0 else root + 12
                track.append(Message('note_on', note=note, velocity=80, time=current_time))
                track.append(Message('note_off', note=note, velocity=80, time=int(mid.ticks_per_beat)))
                current_time = 0

    track.append(mido.MetaMessage('end_of_track', time=0))
    mid.tracks.append(track)

def generate_drum_track(mid):
    """强力打击乐"""
    track = MidiTrack()
    track.name = "drum"

    current_time = 0
    for i in range(TOTAL_BARS * 4):  # 每小节4拍
        # 底鼓（强拍）
        track.append(Message('note_on', note=36, velocity=100, time=current_time))
        track.append(Message('note_off', note=36, velocity=100, time=int(mid.ticks_per_beat/2)))
        current_time = 0

        # 军鼓（弱拍）
        track.append(Message('note_on', note=38, velocity=90, time=int(mid.ticks_per_beat/2)))
        track.append(Message('note_off', note=38, velocity=90, time=int(mid.ticks_per_beat/2)))
        current_time = 0

        # 踩镲（1/8音符）
        for _ in range(2):
            track.append(Message('note_on', note=42, velocity=80, time=current_time))
            track.append(Message('note_off', note=42, velocity=80, time=int(mid.ticks_per_beat/4)))
            current_time = 0

    track.append(mido.MetaMessage('end_of_track', time=0))
    mid.tracks.append(track)

def create_song(filename):
    mid = MidiFile(ticks_per_beat=480)

    # 生成和弦进行（I-V-vi-IV）
    chord_prog = generate_chord_progression()

    # 生成各声部
    generate_melody_track(mid)
    generate_chord_track(mid, chord_prog)
    generate_bass_track(mid, chord_prog)
    generate_drum_track(mid)

    # 添加全局BPM设置
    for track in mid.tracks:
        track.insert(0, mido.MetaMessage('set_tempo', tempo=mido.bpm2tempo(BPM)))

    mid.save(filename)
    return mid

if __name__ == "__main__":
    output = "fast_song.mid"
    create_song(output)
    print(f"2分钟快节奏音乐已生成：{output}")

import random

# 定义多个音阶选项
scales = {

chord_progression = [
    [60, 64, 67],    # C大三和弦（C4-E4-G4）
    [57, 60, 64],    # Am小三和弦（A3-C4-E4）
    [65, 69, 72],    # F大三和弦（F4-A4-C5）
    [67, 71, 74]     # G大三和弦（G4-B4-D5）
]

# 定义可能的节奏型（每个数字代表音符持续时间，单位为拍）
rhythm_patterns = [
    [0.5, 0.5, 0.5, 0.5],           # 均匀的八分音符
    [0.75, 0.25, 0.5, 0.5],         # 附点八分音符 + 十六分音符 + 八分音符
    [1.0, 0.5, 0.5],                # 四分音符 + 八分音符
    [0.25, 0.25, 0.5, 0.5, 0.5]     # 十六分音符 + 混合
]

def generate_melody(start_time, bars, scale, is_chorus=False):
    section_notes = []
    for bar in range(bars):
        current_chord = chord_progression[bar % 4]
        bar_start = start_time + bar * 2.0
        
        # 添加和弦，副歌部分和弦力度增强
        for pitch in current_chord:
            chord_note = pretty_midi.Note(
                velocity=70 if not is_chorus else 80,
                pitch=pitch,
                start=bar_start,
                end=bar_start + 2.0
            )
            section_notes.append(chord_note)
        
        # 选择节奏型，副歌使用更活跃的节奏
        if is_chorus:
            rhythm = rhythm_patterns[2:4]  # 使用更活跃的节奏
        else:
            rhythm = rhythm_patterns[0:2]  # 使用较平稳的节奏
        current_rhythm = random.choice(rhythm)
        
        current_time = bar_start
        for duration in current_rhythm:
            if current_time - bar_start >= 2.0:
                break
            
            # 选择音符，副歌音域略高
            current_pitch = random.choice(scale)
            if is_chorus:
                current_pitch += 12  # 副歌升高八度
            
            # 避免大跳
            if len(section_notes) > 0:
                last_pitch = section_notes[-1].pitch
                attempts = 0
                while abs(current_pitch - last_pitch) > 4 and attempts < 5:
                    current_pitch = random.choice(scale)
                    if is_chorus:
                        current_pitch += 12
                    attempts += 1
            
            note = pretty_midi.Note(
                velocity=random.randint(85, 95) if is_chorus else random.randint(75, 85),
                pitch=current_pitch,
                start=current_time,
                end=current_time + duration * 0.95
            )
            section_notes.append(note)
            current_time += duration
    
    return section_notes

# 创建MIDI对象和音轨

piano = pretty_midi.Instrument(program=pretty_midi.instrument_name_to_program('Acoustic Grand Piano'))

enhanced_music_generator_fixed.py
- 使用正确的乐器名称

import random
from music21 import stream, note, chord, instrument, meter, tempo, dynamics, scale

# 全局配置
BPM = 120
BARS = 8
BEATS_PER_BAR = 4

class EnhancedMusicGenerator:
    def __init__(self):
        self.score = stream.Score()
        self.score.append(meter.TimeSignature('4/4'))
        self.score.append(tempo.MetronomeMark(number=BPM))
        self.setup_instruments()
        self.key = 'C4'

    def setup_instruments(self):
        """使用正确且存在的乐器类"""
        # 主旋律 - 长笛（更柔和的音色）
        self.melody_part = stream.Part()
        self.melody_part.insert(0, instrument.Flute())

        # 和声层 - 弦乐合奏（正确类名为StringEnsemble）
        self.harmony_part = stream.Part()
        self.harmony_part.insert(0, instrument.StringEnsemble())

        # 打击乐 - 使用DrumSet替代DrumKit
        self.drum_part = stream.Part()
        self.drum_part.insert(0, instrument.DrumSet())
        self.drum_part.midiChannel = 9  # MIDI通道10

    def generate_melody(self):
        """改进的旋律生成算法"""
        pentatonic = [
            note.Note(pitch) for pitch in
            scale.MajorScale(self.key).getPitches('C4', 'C5')[0:7:2]  # 五声音阶
        ]

        current_pitch = pentatonic[0]
        for _ in range(BARS * 4):  # 八分音符密度
            # 限制音程跳跃
            next_choices = [
                n for n in pentatonic
                if abs(n.pitch.midi - current_pitch.midi) <= 5
            ]
            current_pitch = random.choice(next_choices)

            # 添加随机装饰音
            if random.random() < 0.2:
                self.melody_part.append(note.Note(
                    current_pitch,
                    quarterLength=0.25
                ))
                self.melody_part.append(note.Note(
                    current_pitch.transpose(2),
                    quarterLength=0.25
                ))
            else:
                self.melody_part.append(note.Note(
                    current_pitch,
                    quarterLength=0.5
                ))

    def generate_harmony(self):
        """经典和弦进行"""
        progression = [
            ('C4', 'E4', 'G4'),  # C大三和弦
            ('F4', 'A4', 'C5'),  # F大三和弦
            ('G4', 'B4', 'D5'),  # G大三和弦
            ('C4', 'E4', 'G4')
        ] * 2

        for chord_notes in progression:
            ch = chord.Chord(chord_notes)
            ch.quarterLength = 2  # 每个和弦持续2拍
            self.harmony_part.append(ch)

    def generate_drums(self):
        """改进的打击乐生成"""
        # 标准MIDI打击乐映射
        DRUM_MAP = {
            'kick': 36,
            'snare': 38,
            'hihat_closed': 42,
            'hihat_open': 46
        }

        # 专业鼓点模式
        pattern = [
            ('kick', 0),      # 第1拍
            ('hihat_closed', 0.5),
            ('snare', 1),      # 第2拍
            ('hihat_closed', 1.5),
            ('kick', 2),       # 第3拍
            ('hihat_closed', 2.5),
            ('snare', 3),      # 第4拍
            ('hihat_closed', 3.5)
        ] * BARS

        for step in pattern:
            drum_type, offset = step
            n = note.Unpitched()
            n.offset = offset
            n.quarterLength = 0.5
            n.midiPitch = DRUM_MAP[drum_type]
            self.drum_part.append(n)

    def export_midi(self, filename):
        """添加动态表情符号"""
        # 旋律动态变化
        self.melody_part.insert(0, dynamics.Diminuendo())
        self.melody_part.insert(4, dynamics.Crescendo())

        # 合并音轨并导出
        self.score.append([self.melody_part, self.harmony_part, self.drum_part])
        self.score.write('midi', fp=filename)

if __name__ == "__main__":
    gen = EnhancedMusicGenerator()
    gen.generate_melody()
    gen.generate_harmony()
    gen.generate_drums()
    gen.export_midi('professional_song.mid')
    print("专业级音乐生成完成！")

            (38, 1.0),    # 军鼓

        ] * (BARS * 2)

        for pitch, offset in pattern:
            n = note.Unpitched()
            n.midiPitch = pitch
            n.offset = offset
            n.quarterLength = 0.5
            self.drum_part.append(n)

    def export_midi(self, filename):
        """导出MIDI文件"""
        self.score.append([self.melody_part, self.harmony_part, self.drum_part])
        self.score.write('midi', fp=filename)
