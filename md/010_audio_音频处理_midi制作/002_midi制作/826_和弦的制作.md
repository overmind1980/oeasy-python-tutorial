mid = MidiFile()

for track_data in tracks_data:
    track = MidiTrack()
    mid.tracks.append(track)

    # 设置速度为96BPM
    track.append(mido.MetaMessage("set_tempo", tempo=625000, time=0))

    # 为打击乐轨道设置通道10（MIDI标准打击乐通道）
    is_drum_track = track_data["track_name"] == "Drums"

    notes = track_data["notes"]
    notes.sort(key=lambda x: x[0])

    current_time = 0
    for start, duration, pitch, velocity in notes:
        wait_time = start - current_time
        if wait_time < 0:
            wait_time = 0

        if is_drum_track:
            track.append(
                Message(
                    "note_on",
                    note=pitch,
                    velocity=velocity,
                    time=wait_time,
                    channel=9,
                )
            )
            track.append(
                Message("note_off", note=pitch, velocity=0, time=duration, channel=9)
            )
        else:
            track.append(
                Message("note_on", note=pitch, velocity=velocity, time=wait_time)
            )
            track.append(Message("note_off", note=pitch, velocity=0, time=duration))

        current_time = start + duration

# 保存MIDI文件

def create_chorus():

    for bar in range(bars // 2):
        chord = chord_progressions[(bar + len(chord_progressions) // 2) % len(chord_progressions)]
        # 生成和弦部分

        # 生成贝斯部分
        bass_note = chord[0] - 12  # 取和弦根音的低八度作为贝斯音

        bass_track.append(mido.Message('note_off', note=bass_note, velocity=random.randint(70, 90), time=ticks_per_beat * beats_per_bar))

        for beat in range(beats_per_bar):

            possible_notes = chord + [note for note in major_scale if note not in chord]
            if previous_note is not None:
