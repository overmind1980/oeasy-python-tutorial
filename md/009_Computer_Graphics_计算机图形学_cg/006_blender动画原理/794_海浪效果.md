import bpy
import math
import random

# ============= 清除旧内容 =============
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# ============= 创建基础场景（天空+光源） =============
def setup_scene():
    # 1. 添加天空背景（模拟环境光）
    world = bpy.context.scene.world
    world.use_nodes = True
    bg_node = world.node_tree.nodes["Background"]
    # 天空颜色：从浅蓝到深蓝渐变
    bg_node.inputs[0].default_value = (0.1, 0.3, 0.8, 1)  # 天空色
    bg_node.inputs[1].default_value = 0.5  # 强度
    
    # 2. 添加主光源（模拟太阳）
    bpy.ops.object.light_add(type='SUN', location=(10, 5, 15))
    sun = bpy.context.active_object
    sun.data.energy = 2.0  # 光照强度
    sun.rotation_euler = (math.radians(60), 0, math.radians(30))  # 光照角度

# ============= 创建海浪网格 =============
def create_ocean_mesh():
    width = 20.0  
    height = 20.0  
    subdivisions_x = 200  # 更高细分，波浪更光滑
    subdivisions_y = 200  

    mesh = bpy.data.meshes.new("OceanMesh")
    obj = bpy.data.objects.new("OceanObject", mesh)
    bpy.context.collection.objects.link(obj)

    verts = []
    faces = []
    for y in range(subdivisions_y + 1):
        for x in range(subdivisions_x + 1):
            vx = x * width / subdivisions_x - width / 2
            vy = y * height / subdivisions_y - height / 2
            vz = 0.0
            verts.append((vx, vy, vz))
    for y in range(subdivisions_y):
        for x in range(subdivisions_x):
            i = x + y * (subdivisions_x + 1)
            face = (i, i + 1, i + subdivisions_x + 2, i + subdivisions_x + 1)
            faces.append(face)
    
    mesh.from_pydata(verts, [], faces)
    mesh.update()
    return obj

# ============= 添加摄像机并对准海浪 =============
def add_camera():
    # 摄像机位置：斜上方，距离海浪适中，确保完整拍摄
    bpy.ops.object.camera_add(location=(12, -12, 6))
    camera = bpy.context.active_object
    # 摄像机朝向：对准海浪中心（0,0,0）
    camera.rotation_euler = (math.radians(50), 0, math.radians(45))  # 角度调整
    # 设置为场景默认摄像机
    bpy.context.scene.camera = camera
    return camera

# ============= 动态更新函数 =============
def update_wave_dynamic(scene):
    obj = ocean_obj
    if not obj:
        return
    
    mesh = obj.data
    time = scene.frame_current  

    amplitude = 1.5  
    speed = 0.3      
    for vertex in mesh.vertices:
        x, y, z = vertex.co
        wave1 = amplitude * math.sin(1.5 * math.pi * (x + speed * time) + 2 * math.pi * y)
        wave2 = amplitude * 0.3 * math.sin(3 * math.pi * (x + speed * time) + 1.5 * math.pi * y + random.random())
        vertex.co.z = wave1 + wave2
    mesh.update()

# ============= 执行场景搭建 =============
setup_scene()  # 创建天空和光源
ocean_obj = create_ocean_mesh()  # 创建海浪
camera = add_camera()  # 添加摄像机

# ============= 注册帧更新 =============
for func in bpy.app.handlers.frame_change_pre:
    if func.__name__ == update_wave_dynamic.__name__:
        bpy.app.handlers.frame_change_pre.remove(func)
bpy.app.handlers.frame_change_pre.append(update_wave_dynamic)

# ============= 手动触发首帧 =============
update_wave_dynamic(bpy.context.scene)
