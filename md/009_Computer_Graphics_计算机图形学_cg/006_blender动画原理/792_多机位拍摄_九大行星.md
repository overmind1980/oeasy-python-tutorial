import bpy

def create_emission_shader(color, strength, name):
    material = bpy.data.materials.new(name)
    material.use_nodes = True
    nodes = material.node_tree.nodes
    links = material.node_tree.links

    # 创建并设置发射节点
    emission = nodes.new('ShaderNodeEmission')
    emission.inputs['Color'].default_value = color
    emission.inputs['Strength'].default_value = strength

    # 创建并设置输出节点
    output = nodes.new('ShaderNodeOutputMaterial')

    # 连接节点
    links.new(emission.outputs['Emission'], output.inputs['Surface'])

    return material

def create_sphere(radius, location, name):  
    # 取消选择所有对象，以确保新创建的对象是唯一被选中的  
    bpy.ops.object.select_all(action='DESELECT')  
      
    # 创建 UV 球体  
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=location)  
      
    # 新创建的对象会被自动选中，我们可以获取这个对象并设置其名称  
    sphere_obj = bpy.context.selected_objects[0]  
    sphere_obj.name = name  
    return sphere_obj


def create_torus(location, name):  
    # 取消选择所有对象，以确保新创建的对象是唯一被选中的  
    bpy.ops.object.select_all(action='DESELECT')  
      
    # 创建圆环体（Torus）  
    bpy.ops.mesh.primitive_torus_add(location=location)  
      
    # 新创建的对象会被自动选中，我们可以获取这个对象并设置其名称  
    torus_obj = bpy.context.selected_objects[0]  
    torus_obj.name = name  
      
    # 返回新创建并命名的圆环体对象  
    return torus_obj 


def render_camera_to_mp4(camera_obj, output_folder=None, output_filename_prefix="animation"): 
    # 根据操作系统获取桌面路径  
    if platform.system() == "Windows":  
        desktop_path = os.path.join(os.path.expanduser("~"), "Desktop", "my_animation_frames")  
    elif platform.system() == "Darwin":  
        desktop_path = os.path.expanduser("/tmp/my_animation_frames")  
    else:  
        print("未知系统...")  
        return  
      
    # 如果提供了自定义的输出文件夹，则使用它  
    if output_folder:  
        desktop_path = os.path.expanduser(output_folder)  
      
    # 确保输出文件夹存在  
    if not os.path.exists(desktop_path):  
        os.makedirs(desktop_path)  
      
    # 设置输出格式为FFmpeg视频  
    bpy.context.scene.render.engine = 'CYCLES'  # 或者 'BLENDER_EEVEE'，取决于你想要的渲染效果  
    bpy.context.scene.render.image_settings.file_format = 'FFMPEG'  
  
    # FFmpeg设置（这里使用H.264编码）  
    bpy.context.scene.render.ffmpeg.format = 'MPEG4'  
    bpy.context.scene.render.ffmpeg.codec = 'H264'  
  
    # 设置输出文件路径和文件名  
    bpy.context.scene.camera = camera_obj  # 设置当前摄像机  
    output_filename = f"{output_filename_prefix}_{camera_obj.name}.mp4"
    print(output_filename)
    bpy.context.scene.render.filepath = os.path.join(desktop_path, output_filename)  
      
    # 渲染动画  
    bpy.ops.render.render(animation=True, write_still=False)

def add_light(light_type='SPOT', radius=200, location=(0, 0, 0), energy=10000000, direction=(0, 0, -1)):  
    bpy.ops.object.light_add(type=light_type, location=location)  
    light = bpy.context.object  
    light.data.type = light_type  # 明确设置灯光类型  
    light.data.energy = energy  # 设置灯光能量  
  
    # 根据灯光类型设置半径或类似属性  
    if light_type == 'AREA':  
        light.data.size = radius  # 区域光的尺寸  
    elif light_type in ('SPOT', 'POINT'):  
        light.data.shadow_soft_size = radius  # 聚光灯和点光源使用shadow_soft_size模拟半径  
    # 设置灯光方向  
    if light_type == 'SPOT':  
        # 将方向向量转换为欧拉角  
        direction_vector = mathutils.Vector(direction)  
        rotation_euler = direction_vector.to_track_quat('Z', 'Y').to_euler()  
        light.rotation_euler = rotation_euler  # 设置旋转以指向特定方向  
    return light
