from time import sleep
from typing import Any

def xinput(type, prompt: object = "", invialion_prompt:str = "Invial Data") -> any:
    while 1:
        try:
            return type(input(prompt))
        except Exception:
            print(invialion_prompt)
            continue

def end():
    input(f"##############################\n输入Enter退出\n")
    exit()

class Pet():
    def __init__(self, name, characteristic, feeding_habits_type) -> None:
        self.name = name
        self.characteristic = characteristic
        print(f"这是一只{self.characteristic}的{self.name}")
        self.satiety = 50 # 饱食度
        self.affinity = 50 # 关系
        self.cleanliness = 50 # 洁净度
        self.fatigue = 20 # 劳累
        self.feeding_habits_type = feeding_habits_type
    
    def play(self) -> None:
        print("玩耍中...")
        sleep(2)
        print("玩耍结束，亲和度+10,饱食度-5,整洁度-10,疲劳度+10")
        self.satiety -= 5 # 饱食度
        self.affinity += 10 # 关系
        self.cleanliness -= 10 # 洁净度
        self.fatigue += 10 # 劳累
        if self.affinity > 100:
            self.affinity = 100
    
    def have_a_bath(self) -> None:
        print("洗澡中...")
        sleep(2)
        print('洗澡完成')
        print('饱食度-5,整洁度+50,亲和度-1')
        self.satiety -= 5 # 饱食度
        self.affinity -= 1 # 关系
        self.cleanliness += 50 # 洁净度
        # self.fatigue += 10 # 劳累

    def feeding(self, food_type) -> None:
        if food_type == 1:
            print('宠物粮喂食成功')
            print('饱食度+3,亲和度-2')
            self.satiety += 3
            self.affinity -= 2
        elif food_type == 2:
            if self.feeding_habits_type == 1:
                print('鸡腿喂食成功')
                print('饱食度+5,亲和度+3,整洁度-4')
                self.satiety += 5
                self.affinity -= 3
                self.cleanliness -= 4
            else:
                print('你喂素食动物吃了鸡腿')
                print('饱食度-5,亲和度-2,整洁度-6')
                self.satiety -= 5
                self.affinity -= 2
                self.cleanliness -= 6
        elif food_type == 3:
            if feeding_habits_type == 2:
                print('树叶喂食成功')
                print('饱食度+4,亲和度+3')
                self.satiety += 4
                self.affinity += 3
            else:
                print('你喂食了肉食动物树叶')
                print('饱食度-7,亲和度-3')
                self.satiety -= 7
                self.affinity -= 3
        elif food_type == 4:
            for i in range(100):
                print('你喂你的宠物吃了毒药!'*10)
            print("现在它必死无疑")
            print('再见')
            end()
    
    def check_attrib(self) -> None:
        if self.satiety <= 0:
            print('饱食度低于0,你的宠物饿死了')
            end()
        elif self.satiety >= 150:
            print('饱食度高于150,你的宠物撑死了')
            end()
        elif self.affinity <= -10:
            print('亲和度低于-10,你的宠物离开了你')
            end()
        elif self.cleanliness <= 0:
            print('整洁度低于0,你的宠物被流放')
            end()
        elif self.fatigue >= 150:
            print('疲劳度高于150,你的宠物累死了')
            end()
