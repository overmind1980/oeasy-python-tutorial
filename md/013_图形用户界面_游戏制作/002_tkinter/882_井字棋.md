```
import tkinter as tk
from tkinter import filedialog, messagebox
import pygame
from pygame import mixer
import os
import sys


class MusicPlayer:
    def __init__(self, root):
        self.root = root
        self.setup_window()
        self.initialize_audio()
        self.setup_variables()
        self.create_widgets()
        self.create_menu()

    def setup_window(self):
        self.root.title("音乐播放器")
        self.root.geometry("520x320")
        self.root.resizable(False, False)
        self.root.configure(bg="#f0f0f0")

        # 设置窗口图标（可选）
        try:
            self.root.iconbitmap(self.resource_path("music_icon.ico"))
        except:
            pass

    def resource_path(self, relative_path):
        """ 解决跨平台资源路径问题 """
        try:
            base_path = sys._MEIPASS
        except Exception:
            base_path = os.path.abspath(".")
        return os.path.join(base_path, relative_path)

    def initialize_audio(self):
        """ 带错误处理的音频初始化 """
        try:
            pygame.init()
            mixer.init()
        except Exception as e:
            messagebox.showerror("初始化失败",
                                 f"无法初始化音频设备:\n{str(e)}\n\n"
                                 "请确保：\n1. 已安装声卡驱动\n2. 未占用音频设备\n3. 安装正确依赖库")
            self.root.destroy()
            sys.exit(1)

    def setup_variables(self):
        self.current_file = ""
        self.is_playing = False
        self.is_paused = False
        self.song_length = 0
        self.update_id = None

    def create_widgets(self):
        # 主容器
        main_frame = tk.Frame(self.root, bg="#f0f0f0")
        main_frame.pack(padx=10, pady=10, fill=tk.BOTH, expand=True)

        # 当前歌曲显示
        self.song_label = tk.Label(main_frame,
                                   text="未选择歌曲",
                                   font=("微软雅黑", 12),
                                   bg="#f0f0f0",
                                   fg="#333",
                                   wraplength=450)
        self.song_label.pack(pady=10)

        # 时间显示
        self.time_label = tk.Label(main_frame,
                                   text="00:00 / 00:00",
                                   font=("Consolas", 10),
                                   bg="#f0f0f0",
                                   fg="#666")
        self.time_label.pack()

        # 进度条
        self.progress = tk.Scale(main_frame,
                                 from_=0,
                                 to=100,
                                 orient=tk.HORIZONTAL,
                                 showvalue=False,
                                 sliderrelief=tk.FLAT,
                                 troughcolor="#ddd",
                                 command=self.seek)
        self.progress.pack(fill=tk.X, pady=5)

        # 控制按钮
        btn_frame = tk.Frame(main_frame, bg="#f0f0f0")
        btn_frame.pack(pady=15)

        button_style = {
            "font": ("微软雅黑", 10),
            "width": 8,
            "height": 1,
            "bd": 0,
            "bg": "#4CAF50",
            "fg": "white",
            "activebackground": "#45a049"
        }

        self.play_btn = tk.Button(btn_frame,
                                  text="播放",
                                  command=self.play_pause,
                                  **button_style)
        self.stop_btn = tk.Button(btn_frame,
                                  text="停止",
                                  command=self.stop,
                                  **button_style)
        self.open_btn = tk.Button(btn_frame,
                                  text="打开文件",
                                  command=self.open_file,
                                  bg="#2196F3",
                                  activebackground="#1976D2",
                                  **{k: v for k, v in button_style.items() if k not in ["bg", "activebackground"]})

        self.play_btn.pack(side=tk.LEFT, padx=5)
        self.stop_btn.pack(side=tk.LEFT, padx=5)
        self.open_btn.pack(side=tk.LEFT, padx=5)

        # 音量控制
        vol_frame = tk.Frame(main_frame, bg="#f0f0f0")
        vol_frame.pack(pady=10)

        tk.Label(vol_frame,
                 text="音量:",
                 bg="#f0f0f0",
                 font=("微软雅黑", 9)).pack(side=tk.LEFT)

        self.vol_scale = tk.Scale(vol_frame,
                                  from_=0,
                                  to=100,
                                  orient=tk.HORIZONTAL,
                                  command=self.set_volume,
                                  showvalue=False,
                                  sliderrelief=tk.FLAT,
                                  troughcolor="#ddd",
                                  length=120)
        self.vol_scale.set(70)
        self.vol_scale.pack(side=tk.LEFT)
        mixer.music.set_volume(0.7)

    def create_menu(self):
        menubar = tk.Menu(self.root)
        self.root.config(menu=menubar)

        file_menu = tk.Menu(menubar, tearoff=0)
        file_menu.add_command(label="打开文件", command=self.open_file)
        file_menu.add_separator()
        file_menu.add_command(label="退出", command=self.on_closing)
        menubar.add_cascade(label="文件", menu=file_menu)

    def open_file(self):
        file_types = [
            ("音频文件", "*.mp3 *.wav *.ogg *.flac"),
            ("所有文件", "*.*")
        ]
        file_path = filedialog.askopenfilename(filetypes=file_types)

        if file_path:
            try:
                mixer.music.load(file_path)
                self.current_file = file_path
                self.song_label.config(text=os.path.basename(file_path))
                self.song_length = mixer.Sound(file_path).get_length()
                self.progress.config(to=self.song_length)
                self.update_time_display(0, self.song_length)
                self.play_btn.config(text="播放")
                self.is_playing = False
                self.is_paused = False
            except Exception as e:
                messagebox.showerror("加载失败",
                                     f"无法加载文件：\n{os.path.basename(file_path)}\n\n"
                                     f"错误信息：{str(e)}")
                self.reset_player()

    def play_pause(self):
        if not self.current_file:
            messagebox.showwarning("提示", "请先选择音乐文件！")
            return

        try:
            if not self.is_playing:
                mixer.music.play()
                self.is_playing = True
                self.play_btn.config(text="暂停")
                self.start_progress_update()
            else:
                if not self.is_paused:
                    mixer.music.pause()
                    self.is_paused = True
                    self.play_btn.config(text="继续")
                else:
                    mixer.music.unpause()
                    self.is_paused = False
                    self.play_btn.config(text="暂停")
        except Exception as e:
            messagebox.showerror("播放错误", f"播放失败：\n{str(e)}")
            self.reset_player()

    def stop(self):
        mixer.music.stop()
        self.reset_player()

    def set_volume(self, val):
        mixer.music.set_volume(float(val) / 100)

    def seek(self, pos):
        if self.current_file and mixer.music.get_busy():
            try:
                mixer.music.set_pos(float(pos))
            except:
                pass  # 某些格式不支持定位

    def update_time_display(self, current, total):
        current_min = int(current // 60)
        current_sec = int(current % 60)
        total_min = int(total // 60)
        total_sec = int(total % 60)
        self.time_label.config(
            text=f"{current_min:02}:{current_sec:02} / {total_min:02}:{total_sec:02}"
        )

    def start_progress_update(self):
        if self.is_playing and not self.is_paused:
            try:
                current_time = mixer.music.get_pos() / 1000
                self.progress.set(current_time)
                self.update_time_display(current_time, self.song_length)
                self.update_id = self.root.after(500, self.start_progress_update)
            except:
                self.reset_player()

    def reset_player(self):
        if self.update_id:
            self.root.after_cancel(self.update_id)
        self.is_playing = False
        self.is_paused = False
        self.play_btn.config(text="播放")
        self.progress.set(0)
        self.update_time_display(0, self.song_length)
        self.current_file = ""
        self.song_label.config(text="未选择歌曲")

    def on_closing(self):
        mixer.music.stop()
        self.root.destroy()


if __name__ == "__main__":
    root = tk.Tk()
    try:
        player = MusicPlayer(root)
        root.protocol("WM_DELETE_WINDOW", player.on_closing)
        root.mainloop()
    except Exception as e:
        messagebox.showerror("致命错误", f"程序异常终止：\n{str(e)}")
```
