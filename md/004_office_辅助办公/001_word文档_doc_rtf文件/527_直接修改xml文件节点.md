```
from docx import Document
from docx.shared import RGBColor, Inches
from docx.oxml.ns import qn
from docx.oxml import OxmlElement
from docx.shared import Pt

doc = Document()

table = doc.add_table(rows=2, cols=2)
table.cell(0, 0).text = 'sprint'
table.cell(0, 1).text = 'summer'
table.cell(1, 0).text = 'autumn'
table.cell(1, 1).text = 'winter'

tbl_pr = table._element.find(qn('w:tblPr'))
tbl_borders = OxmlElement('w:tblBorders')
for border_type in ('top', 'left', 'bottom', 'right', 'insideH', 'insideV'):
    border_element = OxmlElement(f'w:{border_type}')
    border_element.set(qn('w:val'), 'single')
    border_element.set(qn('w:space'), '0')
    border_element.set(qn('w:color'), 'auto')
    if border_type.startswith("in"):
        border_element.set(qn('w:sz'), '10')
    else:
        border_element.set(qn('w:sz'), '20')
    tbl_borders.append(border_element)
tbl_pr.append(tbl_borders)
doc.save("4.docx")
```
