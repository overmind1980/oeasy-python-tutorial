from PIL import Image, ImageDraw, ImageFont
import math
import os
from tqdm import tqdm


def calculate_watermark_size(image_size):
    """动态计算水印参数（绿色不透明版）"""
    max_dim = max(image_size)
    base_size = 3000
    ratio = max_dim / base_size

    return {
        'font_size': int(60 * ratio),  # 增大基础字号
        'spacing': int(100 * ratio),  # 减小间距使水印更密集
        'opacity': 255,  # 完全不透明
        'angle': 30
    }


def add_tiled_watermark(image, watermark_text):
    """添加绿色不透明平铺水印"""
    width, height = image.size
    params = calculate_watermark_size((width, height))

    # 原始图片转为RGBA模式
    image = image.convert("RGBA")

    # 创建透明水印层
    watermark = Image.new("RGBA", (width, height), (0, 0, 0, 0))
    draw = ImageDraw.Draw(watermark)

    # 加载字体（建议使用绝对路径）
    try:
        font = ImageFont.truetype("/System/Library/Fonts/Arial.ttf", params['font_size'])
    except:
        font = ImageFont.load_default().font_variant(size=params['font_size'])

    # 获取文本尺寸
    bbox = draw.textbbox((0, 0), watermark_text, font=font)
    text_width = bbox[2] - bbox[0]
    text_height = bbox[3] - bbox[1]
    diagonal = int(math.sqrt(text_width ** 2 + text_height ** 2))

    # 计算平铺行列数
    cols = math.ceil(width / (params['spacing'] + diagonal)) + 2
    rows = math.ceil(height / (params['spacing'] + diagonal)) + 2

    # 创建单个绿色水印单元
    single_watermark = Image.new("RGBA", (text_width, text_height), (0, 0, 0, 0))
    single_draw = ImageDraw.Draw(single_watermark)
    single_draw.text((0, 0), watermark_text, font=font,
                     fill=(0, 255, 0, 255))  # 绿色完全不透明

    # 旋转水印单元
    rotated_watermark = single_watermark.rotate(params['angle'],
                                                expand=True,
                                                resample=Image.BICUBIC)

    # 平铺水印（交错排列）
    for i in range(rows):
        for j in range(cols):
            x = j * (params['spacing'] + diagonal)
            y = i * (params['spacing'] + diagonal)
            if i % 2 == 0:
                x += diagonal // 2
            watermark.paste(rotated_watermark, (x, y), rotated_watermark)

    # 合并图层
    return Image.alpha_composite(image, watermark)


def batch_watermark(input_dir, output_dir, watermark_text):
    """批量处理"""
    os.makedirs(output_dir, exist_ok=True)

    # 支持的文件格式
    extensions = ('.jpg', '.jpeg', '.png', '.webp')
    files = [f for f in os.listdir(input_dir) if f.lower().endswith(extensions)]

    if not files:
        print(f"目录中未找到图片: {input_dir}")
        return

    print(f"正在处理 {len(files)} 张图片...")

    for file in tqdm(files):
        try:
            input_path = os.path.join(input_dir, file)
            output_path = os.path.join(output_dir, file)

            with Image.open(input_path) as img:
                # 添加水印并保存为PNG（保留透明度）
                img = add_tiled_watermark(img, watermark_text)
                img.save(os.path.splitext(output_path)[0] + ".png")

        except Exception as e:
            print(f"\n处理失败: {file} - {e}")


if __name__ == "__main__":
    # 使用示例
    batch_watermark(
        input_dir="/Users/weixiaomeng/Desktop/kite",
        output_dir="/Users/weixiaomeng/Desktop/kite_wm",
        watermark_text="ROSEY"  # 建议使用英文测试（中文需换字体）
    )
